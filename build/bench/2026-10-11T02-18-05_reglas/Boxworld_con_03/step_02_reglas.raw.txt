const Reglas = {
    /**
     * Crea el estado inicial del juego a partir de un nivel dado.
     * @param {string[]} nivel - Array de strings que representan el mapa del nivel (10x10).
     * @returns {object} El estado inicial del juego.
     */
    crearEstado: function(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];
        let nuevoMapa = [];

        // 1. Parsear el nivel para identificar posiciones y construir el mapa base
        for (let r = 0; r < 10; r++) {
            let filaActual = nivel[r].split('');
            nuevoMapa.push(filaActual.join('')); // Almacenamos la representación de la fila

            for (let c = 0; c < 10; c++) {
                const char = nivel[r][c];
                if (char === '@') {
                    jugador = { fila: r, col: c };
                    // El jugador se convierte en espacio vacío en el mapa base
                    filaActual[c] = ' '; 
                } else if (char === '$' || char === '*') {
                    cajas.push({ fila: r, col: c });
                    // Las cajas también se convierten en espacios vacíos para el mapa base
                    filaActual[c] = ' '; 
                } else if (char === '.') {
                    objetivos.push({ fila: r, col: c });
                }
            }
        }

        // 2. Ajustar el mapa final: reemplazar cualquier carácter de objeto móvil con espacio vacío
        let estadoMapa = nuevoMapa.map(fila => fila.split(''));
        for (let r = 0; r < 10; r++) {
            for (let c = 0; c < 10; c++) {
                const char = nivel[r][c];
                if (char === '@' || char === '$' || char === '*') {
                    estadoMapa[r][c] = ' ';
                } else if (char === '.') {
                    // Los objetivos permanecen como '.' en el mapa base, pero deben ser considerados.
                    estadoMapa[r][c] = '.'; 
                }
            }
        }

        return {
            mapa: estadoMapa, // Mapa de caracteres modificable para la representación interna (aunque solo usamos las coordenadas)
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    },

    /**
     * Intenta mover el estado actual en una dirección dada.
     * @param {object} estado - El estado actual del juego.
     * @param {"arriba" | "abajo" | "izquierda" | "derecha"} direccion - Dirección de movimiento.
     * @returns {object} Un nuevo estado o el estado original si el movimiento falla.
     */
    mover: function(estado, direccion) {
        // Crear un estado base para devolverlo en caso de fallo (copia profunda necesaria)
        const nuevoEstado = JSON.parse(JSON.stringify(estado));

        let jugadorPos = { ...estado.jugador };
        let nuevaCajaPos = null; // Posición donde se moverá la caja, si aplica
        let movimientoExitoso = false;

        // 1. Calcular la posición objetivo del jugador (sin considerar cajas)
        let nextPlayerPos = { ...jugadorPos };
        switch (direccion) {
            case "arriba":
                nextPlayerPos.fila -= 1;
                break;
            case "abajo":
                nextPlayerPos.fila += 1;
                break;
            case "izquierda":
                nextPlayerPos.col -= 1;
                break;
            case "derecha":
                nextPlayerPos.col += 1;
                break;
        }

        // Verificar límites del mapa (10x10)
        if (nextPlayerPos.fila < 0 || nextPlayerPos.fila >= 10 || nextPlayerPos.col < 0 || nextPlayerPos.col >= 10) {
            return estado; // Fuera de límites
        }

        // Verificar colisión con pared en la posición objetivo del jugador
        if (estado.mapa[nextPlayerPos.fila][nextPlayerPos.col] === '#') {
            return estado; // Colisión con pared
        }

        // 2. Verificar si hay cajas en el camino o destino
        const cajaEnDestino = estado.cajas.some(c => c.fila === nextPlayerPos.fila && c.col === nextPlayerPos.col);

        if (!cajaEnDestino) {
            // Caso A: Movimiento libre (sin cajas en el camino)
            nuevoEstado.jugador = nextPlayerPos;
            nuevoEstado.movimientos += 1;
            return nuevoEstado;
        } else {
            // Caso B: Colisión con caja, intentar empujar
            let cajaAEmpujar = estado.cajas.find(c => c.fila === nextPlayerPos.fila && c.col === nextPlayerPos.col);

            // Calcular la posición objetivo de la caja (detrás del jugador)
            let boxNextPos = { ...nextPlayerPos };
            switch (direccion) {
                case "arriba":
                    boxNextPos.fila -= 1;
                    break;
                case "abajo":
                    boxNextPos.fila += 1;
                    break;
                case "izquierda":
                    boxNextPos.col -= 1;
                    break;
                case "derecha":
                    boxNextPos.col += 1;
                    break;
            }

            // Verificar si el empuje es posible:
            // 1. No puede salir del mapa (límites)
            if (boxNextPos.fila < 0 || boxNextPos.fila >= 10 || boxNextPos.col < 0 || boxNextPos.col >= 10) {
                return estado; // Empuje fuera de límites
            }

            // 2. No puede chocar con una pared en el destino de la caja
            if (estado.mapa[boxNextPos.fila][boxNextPos.col] === '#') {
                return estado; // Empuje contra pared
            }

            // 3. No puede chocar con otra caja en el destino de la caja
            const hayOtraCaja = estado.cajas.some(c => c !== cajaAEmpujar && c.fila === boxNextPos.fila && c.col === boxNextPos.col);
            if (hayOtraCaja) {
                return estado; // Empuje contra otra caja
            }

            // Si todo es válido, el movimiento ocurre:
            nuevoEstado.jugador = nextPlayerPos;
            nuevoEstado.movimientos += 1;

            // Actualizar la posición de la caja empujada
            const nuevasCajas = estado.cajas.map(c => {
                if (c === cajaAEmpujar) {
                    return boxNextPos; // La caja se mueve a boxNextPos
                }
                return c; // Otras cajas permanecen en su lugar
            });
            nuevoEstado.cajas = nuevasCajas;

            return nuevoEstado;
        }
    },

    /**
     * Verifica si el nivel ha sido completado (todas las cajas están sobre objetivos).
     * @param {object} estado - El estado actual del juego.
     * @returns {boolean} True si todas las cajas cubren un objetivo, False en caso contrario.
     */
    ganado: function(estado) {
        if (!estado.objetivos || estado.objetivos.length === 0) {
            return true; // No hay objetivos que cubrir
        }

        // Creamos un set de coordenadas de los objetivos para una búsqueda rápida
        const objetivosSet = new Set(estado.objetivos.map(obj => `${obj.fila},${obj.col}`));

        let cajasSobreObjetivoCount = 0;
        let totalCajas = estado.cajas.length;

        // Contar cuántas cajas están sobre un objetivo
        for (const caja of estado.cajas) {
            if (objetivosSet.has(`${caja.fila},${caja.col}`)) {
                cajasSobreObjetivoCount++;
            }
        }

        // El nivel está ganado si el número de cajas es igual al número de objetivos, 
        // y cada caja está sobre un objetivo único (asumiendo que no hay más cajas que objetivos).
        // La condición estricta es: todas las cajas deben estar en una posición que sea también un objetivo.
        return totalCajas === estado.objetivos.length && cajasSobreObjetivoCount === totalCajas;
    }
};
