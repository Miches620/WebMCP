crearEstado: function(nivel) {
        let jugadorPos = null;
        let cajasIniciales = [];
        let objetivosIniciales = [];
        const nuevoMapaArray = [];

        for (let r = 0; r < nivel.length; r++) {
            let newRowChars = [];
            for (let c = 0; c < nivel[r].length; c++) {
                const char = nivel[r][c];
                let mapChar = '';

                // 1. Detect state changes based on original character
                if (char === '@' || char === '+') { // Player start marker
                    if (!jugadorPos) jugadorPos = { fila: r, col: c };
                }
                if (char === '$' || char === '*') { // Box markers
                    cajasIniciales.push({ fila: r, col: c });
                }
                // Objectives are '.', '+', '*'
                if (char === '.' || char === '+' || char === '*') { 
                    objetivosIniciales.push({ fila: r, col: c });
                }

                // 2. Determine the character for the clean map representation
                if (char === '#') {
                    mapChar = '#';
                } else if (char === ' ') {
                    mapChar = ' ';
                } else {
                    // All markers (@, $, *, +, .) become '.' in the clean map.
                    mapChar = '.';
                }
                newRowChars.push(mapChar);
            }
            nuevoMapaArray.push(newRowChars.join(''));
        }

        return {
            mapa: nuevoMapaArray, 
            jugador: jugadorPos,
            cajas: cajasIniciales,
            objetivos: objetivosIniciales,
            movimientos: 0
        };
    },
