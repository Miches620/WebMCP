crearEstado: function(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];
        let mapaBase = [];

        // 1. Parsear el nivel para encontrar posiciones y construir el mapa base
        for (let r = 0; r < 10; r++) {
            if (!nivel[r]) continue; // Asegurarse de que la fila exista
            const rowString = nivel[r];
            let filaMapa = [];

            for (let c = 0; c < 10; c++) {
                // Usar indexación de string para evitar errores con split()
                if (c >= rowString.length) break; 
                const char = rowString[c];

                if (char === '@') {
                    jugador = { r: r, c: c };
                    filaMapa.push(' '); // El jugador no se dibuja en el mapa base
                } else if (char === '$' || char === '*') {
                    // Maneja $ y * como cajas
                    cajas.push({ r: r, c: c });
                    filaMapa.push(' '); // Las cajas no se dibujan en el mapa base
                } else if (char === '.') {
                    objetivos.push({ r: r, c: c });
                    filaMapa.push('.');
                } else if (char === '#') {
                    filaMapa.push('#');
                } else {
                    // Espacio vacío o cualquier otro carácter que no sea especial (+, etc.)
                    filaMapa.push(' ');
                }
            }
            mapaBase.push(filaMapa);
        }

        return {
            mapa: mapaBase,
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    },
