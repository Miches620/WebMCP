crearEstado: function(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];
        let mapaBase = [];

        // 1. Parsear el nivel para encontrar posiciones y construir el mapa base
        for (let r = 0; r < nivel.length; r++) {
            let filaActual = nivel[r].split('');
            let nuevaFila = [];
            for (let c = 0; c < filaActual.length; c++) {
                const char = filaActual[c];
                let mapChar = '';

                if (char === '#') {
                    mapChar = '#';
                } else if (char === '.') {
                    // Objetivo
                    objetivos.push({fila: r, col: c});
                    mapChar = '.';
                } else if (char === '@') {
                    // Jugador
                    jugador = {fila: r, col: c};
                    mapChar = ' '; // El jugador no se dibuja en el mapa base
                } else if (char === '$') {
                    // Caja sobre piso vacío
                    cajas.push({fila: r, col: c});
                    mapChar = ' '; // La caja no se dibuja en el mapa base
                } else if (char === '+') {
                    // Jugador sobre objetivo
                    jugador = {fila: r, col: c};
                    objetivos.push({fila: r, col: c});
                    mapChar = ' '; // El jugador no se dibuja en el mapa base
                } else if (char === '*') {
                    // Caja sobre objetivo
                    cajas.push({fila: r, col: c});
                    objetivos.push({fila: r, col: c});
                    mapChar = '.'; // La caja está sobre un objetivo, por lo que el mapa muestra '.'
                } else {
                    // Espacio vacío ' ' o cualquier otro carácter de piso
                    mapChar = ' ';
                }

                nuevaFila.push(mapChar);
            }
            mapaBase.push(nuevaFila.join(''));
        }

        return {
            mapa: mapaBase, // Solo contiene '#', ' ', '.'
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    },
