crearEstado(nivel) {
    const filas = nivel.length;
    if (filas === 0) return { mapa: [], jugador: null, cajas: [], objetivos: [], movimientos: 0 };
    const columnas = nivel[0].length;

    let jugadorPos = null;
    let cajasIniciales = [];
    let objetivos = [];
    let mapaBase = [];

    // Paso 1: Recorrer el nivel para identificar elementos y construir el mapa base.
    for (let r = 0; r < filas; r++) {
        mapaBase[r] = [];
        const filaStr = nivel[r];
        for (let c = 0; c < columnas; c++) {
            const char = filaStr[c];
            let mapChar = ' '; // Default floor space

            if (char === '#') {
                mapChar = '#';
            } else if (char === '.') {
                mapChar = '.'; // Objetivo/Piso objetivo
                objetivos.push({ fila: r, col: c });
            } else if (char === '@') {
                jugadorPos = { fila: r, col: c };
                // mapChar remains ' ', jugador no se dibuja en el mapa base
            } else if (char === '$' || char === '*') {
                // Las cajas se registran por su posición inicial. El mapa debe mostrar piso.
                cajasIniciales.push({ fila: r, col: c });
                mapChar = ' '; 
            } else if (char === '+') {
                // Jugador sobre objetivo. Muestra el objetivo en el mapa base.
                mapChar = '.';
                objetivos.push({ fila: r, col: c });
            } else {
                // Espacio libre (' ') o cualquier otro carácter de piso.
                mapChar = ' ';
            }

            mapaBase[r][c] = mapChar;
        }
    }

    return {
        mapa: mapaBase,
        jugador: jugadorPos,
        cajas: cajasIniciales,
        objetivos: objetivos,
        movimientos: 0
    };
}
