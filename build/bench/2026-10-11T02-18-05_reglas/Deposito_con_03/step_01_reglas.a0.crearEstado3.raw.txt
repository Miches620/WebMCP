crearEstado(nivel) {
    const mapa = [];
    let jugadorPos = null;
    const cajas = [];
    // Usamos un Set de strings "r,c" para garantizar la unicidad de los objetivos
    const objectivesSet = new Set(); 

    for (let r = 0; r < nivel.length; r++) {
        let rowString = '';
        for (let c = 0; c < nivel[r].length; c++) {
            const char = nivel[r][c];
            let displayChar = ' '; // Valor por defecto: piso vacío

            if (char === '#') {
                displayChar = '#';
            } else if (char === '.') {
                // Objetivo, el mapa muestra '.'
                displayChar = '.'; 
                objectivesSet.add(`${r},${c}`);
            } else if (char === '@' || char === '+') {
                // Jugador inicial: se registra la posición y en el mapa aparece ' '
                if (!jugadorPos) jugadorPos = { fila: r, col: c };
                displayChar = ' '; 
                objectivesSet.add(`${r},${c}`); // '+' es también un objetivo
            } else if (char === '$' || char === '*') {
                // Caja inicial: se registra la posición y en el mapa aparece ' '
                cajas.push({ fila: r, col: c });
                displayChar = ' '; 
                if (char === '*') {
                    objectivesSet.add(`${r},${c}`); // '*' es también un objetivo
                }
            } else if (char === ' ') {
                // Piso vacío
                displayChar = ' ';
            } else {
                // Manejo de cualquier otro carácter inesperado, se trata como piso vacío
                displayChar = ' '; 
            }
            rowString += displayChar;
        }
        mapa.push(rowString); // Añadimos la fila completa como string
    }

    // Convertir el Set de coordenadas únicas a un array de objetos {fila, col}
    const objetivosArray = [];
    for (const key of objectivesSet) {
        const [r, c] = key.split(',');
        objectivosArray.push({ fila: parseInt(r), col: parseInt(c) });
    }

    return {
        mapa: mapa, // Array de strings
        jugador: jugadorPos,
        cajas: cajas,
        objetivos: objetivosArray,
        movimientos: 0
    };
}
