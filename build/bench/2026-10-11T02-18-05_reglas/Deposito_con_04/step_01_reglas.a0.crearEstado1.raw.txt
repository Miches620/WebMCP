crearEstado(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];
        let mapaBase = [];

        // 1. Recorrer el nivel para identificar elementos y construir el mapa base
        for (let r = 0; r < nivel.length; r++) {
            let filaActual = [];
            for (let c = 0; c < nivel[r].length; c++) {
                const char = nivel[r][c];

                if (char === '#') {
                    filaActual.push('#'); // Pared
                } else if (char === '.') {
                    filaActual.push('.'); // Objetivo
                    objetivos.push({ fila: r, col: c });
                } else if (char === ' ') {
                    filaActual.push(' '); // Piso vacío
                } else if (char === '@') {
                    // Jugador inicial
                    jugador = { fila: r, col: c };
                    filaActual.push(' '); // El mapa base no muestra al jugador
                } else if (char === '$' || char === '*') {
                    // Caja ($ o * sobre objetivo)
                    cajas.push({ fila: r, col: c });
                    filaActual.push(' '); // El mapa base no muestra las cajas
                } else if (char === '+') {
                    // Jugador sobre objetivo (+)
                    jugador = { fila: r, col: c };
                    filaActual.push(' '); 
                } else {
                    // Manejar cualquier otro carácter que pueda aparecer en el nivel de entrada
                    filaActual.push(' '); 
                }
            }
            mapaBase.push(filaActual.join(''));
        }

        return {
            mapa: mapaBase,
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    }
