crearEstado(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];
        let mapaBase = [];

        // 1. Recorrer el nivel para identificar elementos y construir el mapa base
        for (let r = 0; r < nivel.length; r++) {
            let filaActual = [];
            for (let c = 0; c < nivel[r].length; c++) {
                const char = nivel[r][c];
                let mapChar = ' '; // Default floor

                // Manejar Paredes
                if (char === '#') {
                    mapChar = '#';
                } 
                // Manejar Objetivos/Marcadores: ., +, *
                else if (char === '.' || char === '+' || char === '*') {
                    objetivos.push({ fila: r, col: c });
                    mapChar = '.'; // El mapa siempre muestra el objetivo como '.'
                }
                // Manejar Jugador Inicial: @ o +
                else if (char === '@' || char === '+') {
                    if (char === '@') {
                        jugador = { fila: r, col: c };
                    } else if (char === '+') {
                        // Si es '+' está sobre un objetivo y es el jugador inicial
                        jugador = { fila: r, col: c };
                    }
                    mapChar = ' '; // La posición del jugador no se muestra en el mapa base
                }
                // Manejar Cajas: $ o *
                else if (char === '$' || char === '*') {
                    cajas.push({ fila: r, col: c });
                    mapChar = ' '; // La posición de la caja no se muestra en el mapa base
                } 
                // Manejar Piso vacío
                else if (char === ' ') {
                     mapChar = ' ';
                } else {
                    // Fallback para cualquier otro carácter inesperado
                    mapChar = ' ';
                }

                filaActual.push(mapChar);
            }
            mapaBase.push(filaActual.join(''));
        }

        return {
            mapa: mapaBase,
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    }
