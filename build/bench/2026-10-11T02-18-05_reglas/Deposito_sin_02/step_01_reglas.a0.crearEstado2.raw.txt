crearEstado: function(nivel) {
    let jugador = null;
    let cajas = [];
    // Usamos un Set para asegurar que cada objetivo se registre solo una vez, independientemente del símbolo inicial.
    let objetivosSet = new Set(); 
    let mapaBase = [];

    for (let r = 0; r < nivel.length; r++) {
        let filaActualChars = [];
        for (let c = 0; c < nivel[r].length; c++) {
            const simbolo = nivel[r][c];
            let charToAdd = ' '; // Valor por defecto para el mapa

            if (simbolo === '#') {
                charToAdd = '#';
            } else if (simbolo === '@' || simbolo === '+') {
                // Jugador: Actualizar posición y dejar espacio vacío en el mapa.
                jugador = { fila: r, col: c };
                charToAdd = ' '; 
            } else if (simbolo === '$' || simbolo === '*') {
                // Caja: Registrar caja y dejar espacio vacío en el mapa.
                cajas.push({ fila: r, col: c });
                charToAdd = ' '; 
            } else if (simbolo === '.') {
                // Objetivo puro: Registrar objetivo y mantener marcador de objetivo en el mapa.
                objetivosSet.add(`${r},${c}`);
                charToAdd = '.';
            } else if (simbolo === '+' || simbolo === '*') {
                 // Caso especial: Jugador sobre objetivo (+) o Caja sobre objetivo (*).
                 // 1. Registrar como objetivo.
                 objetivosSet.add(`${r},${c}`);
                 // 2. El mapa debe mostrar el marcador de objetivo, no la caja/jugador.
                 charToAdd = '.';
            } else {
                // Espacio vacío o piso normal (' ')
                charToAdd = ' ';
            }
            filaActualChars.push(charToAdd);
        }
        mapaBase.push(filaActualChars.join('')); // Convertir el array de caracteres a un string
    }

    // Convertir el Set de coordenadas únicas a la estructura de array requerida para objetivos.
    const objetivosArray = Array.from(objetivosSet).map(coord => {
        const [r, c] = coord.split(',');
        return { fila: parseInt(r), col: parseInt(c) };
    });

    return {
        mapa: mapaBase,
        jugador: jugador,
        cajas: cajas,
        objetivos: objetivosArray,
        movimientos: 0
    };
}
