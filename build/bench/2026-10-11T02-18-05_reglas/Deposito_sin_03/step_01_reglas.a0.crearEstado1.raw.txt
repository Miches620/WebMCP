crearEstado: function(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];

        // 1. Parsear el nivel para encontrar posiciones iniciales
        for (let r = 0; r < 8; r++) {
            for (let c = 0; c < 8; c++) {
                const char = nivel[r][c];
                
                // Detectar jugador: @ o +
                if (char === '@' || char === '+') {
                    jugador = { fila: r, col: c };
                } 
                // Detectar cajas: $ o *
                else if (char === '$' || char === '*') {
                    cajas.push({ fila: r, col: c });
                } 
                // Detectar objetivos: .
                else if (char === '.') {
                    objetivos.push({ fila: r, col: c });
                }
            }
        }

        // 2. Crear el mapa base para el estado (solo #, ' ', .)
        let mapa = [];
        for (let r = 0; r < 8; r++) {
            let filaString = '';
            for (let c = 0; c < 8; c++) {
                const char = nivel[r][c];
                if (char === '#') {
                    filaString += '#'; // Pared
                } else if (char === '.') {
                    filaString += '.'; // Objetivo
                } else {
                    // Piso vacío o posición inicial de caja/jugador que se limpia del mapa base.
                    filaString += ' '; 
                }
            }
            mapa.push(filaString);
        }

        return {
            mapa: mapa,
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    },
