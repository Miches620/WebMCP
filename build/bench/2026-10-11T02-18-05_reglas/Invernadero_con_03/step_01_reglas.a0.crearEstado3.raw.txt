crearEstado(nivel) {
    let jugador = null;
    let cajas = [];
    let objetivos = [];
    const R = nivel.length;
    if (R === 0) {
        return { mapa: [], jugador: null, cajas: [], objetivos: [], movimientos: 0 };
    }
    const C = nivel[0].length;

    let mapaLimpioArray = [];

    for (let r = 0; r < R; r++) {
        let filaActual = '';
        for (let c = 0; c < C; c++) {
            const char = nivel[r][c];
            let mapChar = char; // Default assumption for the map

            // 1. Player detection (@ or +)
            if (char === '@' || char === '+') {
                objetivos.push({ fila: r, col: c });
                jugador = { fila: r, col: c };
                mapChar = '.'; // Player location is an objective marker in the map
            } 
            // 2. Box detection ($ or *)
            else if (char === '$' || char === '*') {
                cajas.push({ fila: r, col: c });
                if (char === '*') {
                    objetivos.push({ fila: r, col: c }); // Box on objective counts as an objective
                }
                mapChar = ' '; // Box location is empty floor space in the map
            } 
            // 3. Objective detection (. or * handled above)
            else if (char === '.') {
                objetivos.push({ fila: r, col: c });
                mapChar = '.';
            }
            // 4. Walls/Floor (Default)
            else if (char === '#') {
                mapChar = '#';
            } else if (char === ' ') {
                mapChar = ' ';
            }

            filaActual += mapChar;
        }
        mapaLimpioArray.push(filaActual);
    }


    const estado = {
        mapa: mapaLimpioArray,
        jugador: jugador,
        cajas: cajas,
        objetivos: objetivos,
        movimientos: 0
    };

    return estado;
}
