crearEstado: function(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];
        let nuevoMapa = [];

        for (let r = 0; r < nivel.length; r++) {
            let filaActual = '';
            for (let c = 0; c < nivel[r].length; c++) {
                const char = nivel[r][c];
                let mapChar = ' '; // Default floor space

                // 1. Detect elements and update lists/player position
                if (char === '@') {
                    jugador = { fila: r, col: c };
                } else if (char === '$' || char === '*') {
                    cajas.push({ fila: r, col: c });
                }

                // Objectives are marked by ., +, or *
                if (char === '.' || char === '+' || char === '*') {
                    objetivos.push({ fila: r, col: c });
                }

                // 2. Determine the static map character for this cell
                if (char === '#') {
                    mapChar = '#';
                } else if (char === '.' || char === '+' || char === '*') {
                    // If it's an objective marker, the map shows '.'
                    mapChar = '.';
                } else if (char === '@' || char === '$') {
                    // Player start or box on floor are treated as empty space in the static map
                    mapChar = ' ';
                } else {
                    // Empty space (' ') remains ' '
                    mapChar = ' ';
                }

                filaActual += mapChar;
            }
            nuevoMapa.push(filaActual);
        }

        return {
            mapa: nuevoMapa,
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    },
