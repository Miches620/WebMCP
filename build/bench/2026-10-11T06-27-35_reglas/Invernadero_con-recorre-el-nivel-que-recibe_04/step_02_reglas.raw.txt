const Reglas = {
    /**
     * Crea el estado inicial del juego a partir de un nivel dado.
     * @param {string[]} nivel - Array de strings que representan el mapa.
     * @returns {object} El estado inicial del juego.
     */
    crearEstado: function(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];

        // 1. Recorrer el nivel para encontrar posiciones iniciales
        for (let r = 0; r < nivel.length; r++) {
            for (let c = 0; c < nivel[r].length; c++) {
                const char = nivel[r][c];

                if (char === '@') {
                    jugador = { fila: r, col: c };
                } else if (char === '$' || char === '*') {
                    // Las cajas se almacenan por coordenadas. No importa si es $ o * inicialmente.
                    cajas.push({ fila: r, col: c });
                } else if (char === '.') {
                    objetivos.push({ fila: r, col: c });
                }
            }
        }

        // 2. Crear el mapa limpio (solo #, ' ', '.')
        let mapaLimpio = nivel.map(fila => fila.split('').map(char => {
            if (char === '@' || char === '$' || char === '*') return ' '; // El jugador y las cajas ocupan espacio vacío en el mapa base
            return char;
        }).join(''));

        // 3. Devolver el estado inicial
        return {
            mapa: mapaLimpio,
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    },

    /**
     * Intenta mover al jugador en la dirección dada. Maneja colisiones y empujes de macetas.
     * @param {object} estado - El estado actual del juego.
     * @param {"arriba" | "abajo" | "izquierda" | "derecha"} direccion - Dirección deseada.
     * @returns {object} Un nuevo estado (o el original si no hubo movimiento).
     */
    mover: function(estado, direccion) {
        // Clonar el estado para asegurar inmutabilidad
        let nuevoEstado = JSON.parse(JSON.stringify(estado));

        const { fila: r, col: c } = estado.jugador;
        let dr = 0, dc = 0;

        switch (direccion) {
            case "arriba":
                dr = -1; dc = 0; break;
            case "abajo":
                dr = 1; dc = 0; break;
            case "izquierda":
                dr = 0; dc = -1; break;
            case "derecha":
                dr = 0; dc = 1; break;
        }

        // Posición tentativa del jugador (P_new)
        const rNewPlayer = r + dr;
        const cNewPlayer = c + dc;

        // 1. Validación de límites y paredes en P_new
        if (rNewPlayer < 0 || rNewPlayer >= estado.mapa.length || 
            cNewPlayer < 0 || cNewPlayer >= estado.mapa[0].length || 
            estado.mapa[rNewPlayer][cNewPlayer] === '#') {
            return nuevoEstado; // Colisión con pared o límite
        }

        // Verificar si P_new está ocupado por una caja
        const boxAtPNew = estado.cajas.some(box => 
            box.fila === rNewPlayer && box.col === cNewPlayer
        );

        if (!boxAtPNew) {
            // Caso A: Movimiento simple (a espacio vacío o objetivo)
            nuevoEstado.jugador = { fila: rNewPlayer, col: cNewPlayer };
            nuevoEstado.movimientos += 1;
            return nuevoEstado;
        } else {
            // Caso B: Intento de empujar una caja
            const boxToPush = estado.cajas.find(box => 
                box.fila === rNewPlayer && box.col === cNewPlayer
            );

            // Posición tentativa del destino de la caja (B_target)
            const rTargetBox = rNewPlayer + dr;
            const cTargetBox = cNewPlayer + dc;

            // 2. Validación del destino B_target
            if (rTargetBox < 0 || rTargetBox >= estado.mapa.length || 
                cTargetBox < 0 || cTargetBox >= estado.mapa[0].length) {
                return nuevoEstado; // Colisión con límite
            }

            // Verificar si B_target es pared o está ocupado por otra caja (o el jugador, aunque esto no debería pasar en un movimiento válido)
            if (estado.mapa[rTargetBox][cTargetBox] === '#') {
                return nuevoEstado; // Colisión con pared
            }

            const boxCollision = estado.cajas.some(box => 
                box !== boxToPush && box.fila === rTargetBox && box.col === cTargetBox
            );

            if (boxCollision) {
                return nuevoEstado; // Colisión con otra caja
            }

            // Empuje exitoso: Actualizar estado
            nuevoEstado.jugador = { fila: rNewPlayer, col: cNewPlayer };
            
            // Crear nueva lista de cajas actualizando la posición de boxToPush
            let nuevasCajas = estado.cajas.map(box => 
                (box === boxToPush) ? { fila: rTargetBox, col: cTargetBox } : box
            );
            nuevoEstado.cajas = nuevasCajas;

            nuevoEstado.movimientos += 1;
            return nuevoEstado;
        }
    },

    /**
     * Verifica si todas las macetas están sobre un cantero objetivo.
     * @param {object} estado - El estado actual del juego.
     * @returns {boolean} True si el nivel está ganado, False en caso contrario.
     */
    ganado: function(estado) {
        if (estado.objetivos.length === 0) return true; // No hay objetivos que cumplir

        // Crear un set de coordenadas objetivo para búsqueda rápida
        const objetivosSet = new Set(estado.objetivos.map(obj => `${obj.fila},${obj.col}`));

        // Verificar si cada caja está sobre un objetivo
        let todasCubiertas = estado.cajas.every(box => {
            return objetivosSet.has(`${box.fila},${box.col}`);
        });

        return todasCubiertas;
    }
};
