const Reglas = {
    /**
     * Crea el estado inicial del juego a partir de un nivel dado.
     * @param {string[]} nivel - Array de strings que representan el mapa (filas).
     * @returns {object} El estado inicial del juego.
     */
    crearEstado: function(nivel) {
        const filas = nivel.length;
        if (filas === 0) return null;
        const columnas = nivel[0].length;

        let jugadorInicial = { fila: -1, col: -1 };
        let cajasIniciales = [];
        let objetivosIniciales = [];
        let mapaBase = [];

        // Primera pasada para identificar posiciones y construir el mapa base
        for (let f = 0; f < filas; f++) {
            let filaActual = [];
            for (let c = 0; c < columnas; c++) {
                const char = nivel[f][c];

                if (char === '@') {
                    jugadorInicial = { fila: f, col: c };
                    filaActual.push(' '); // El jugador no se cuenta en el mapa base
                } else if (char === '$') {
                    cajasIniciales.push({ fila: f, col: c });
                    filaActual.push(' '); // La caja no se cuenta en el mapa base
                } else if (char === '.') {
                    objetivosIniciales.push({ fila: f, col: c });
                    filaActual.push('.');
                } else if (char === '#') {
                    filaActual.push('#');
                } else { // ' '
                    filaActual.push(' ');
                }
            }
            mapaBase.push(filaActual);
        }

        return {
            mapa: mapaBase,
            jugador: jugadorInicial,
            cajas: cajasIniciales,
            objetivos: objetivosIniciales,
            movimientos: 0
        };
    },

    /**
     * Intenta mover el estado actual según la dirección. Maneja colisiones y empujes.
     * @param {object} estado - El estado actual del juego.
     * @param {"arriba" | "abajo" | "izquierda" | "derecha"} direccion - Dirección de movimiento.
     * @returns {object} Un nuevo estado (o el original si no hay movimiento).
     */
    mover: function(estado, direccion) {
        // Crear un estado base para devolverlo en caso de fallo o éxito
        let nuevoEstado = JSON.parse(JSON.stringify(estado));
        nuevoEstado.movimientos = estado.movimientos; // Inicialmente no cambia

        const p = estado.jugador;
        let nuevaPosicionJugador = { fila: p.fila, col: p.col };

        // 1. Calcular la posición objetivo del jugador (sin considerar cajas)
        switch (direccion) {
            case "arriba":
                nuevaPosicionJugador.fila -= 1;
                break;
            case "abajo":
                nuevaPosicionJugador.fila += 1;
                break;
            case "izquierda":
                nuevaPosicionJugador.col -= 1;
                break;
            case "derecha":
                nuevaPosicionJugador.col += 1;
                break;
        }

        // Verificar límites y paredes en la posición objetivo del jugador
        const filas = estado.mapa.length;
        const columnas = state.mapa[0].length;

        if (nuevaPosicionJugador.fila < 0 || nuevaPosicionJugador.fila >= filas ||
            nuevaPosicionJugador.col < 0 || nuevaPosicionJugador.col >= columnas) {
            return estado; // Fuera de límites
        }

        // Verificar si la posición objetivo es una pared
        if (estado.mapa[nuevaPosicionJugador.fila][nuevaPosicionJugador.col] === '#') {
            return estado; // Colisión con pared
        }

        // 2. Comprobar colisiones con cajas
        const cajaEnDestino = estado.cajas.some(c => c.fila === nuevaPosicionJugador.fila && c.col === nuevaPosicionJugador.col);

        if (!cajaEnDestino) {
            // Caso 1: Movimiento libre (sin cajas en el destino)
            nuevoEstado.jugador = nuevaPosicionJugador;
            nuevoEstado.movimientos += 1;
            return nuevoEstado;
        } else {
            // Caso 2: Colisión con caja, intentar empujar
            const cajaAEmpujar = estado.cajas.find(c => c.fila === nuevaPosicionJugador.fila && c.col === nuevaPosicionJugador.col);

            // Calcular la posición detrás de la caja (donde queremos que vaya)
            let posicionDetrasCaja = { fila: cajaAEmpujar.fila, col: cajaAEmpujar.col };
            switch (direccion) {
                case "arriba":
                    posicionDetrasCaja.fila -= 1;
                    break;
                case "abajo":
                    posicionDetrasCaja.fila += 1;
                    break;
                case "izquierda":
                    posicionDetrasCaja.col -= 1;
                    break;
                case "derecha":
                    posicionDetrasCaja.col += 1;
                    break;
            }

            // Verificar si el destino del empuje es válido (dentro de límites, no pared)
            const filasMap = estado.mapa.length;
            const columnasMap = state.mapa[0].length;

            if (posicionDetrasCaja.fila < 0 || posicionDetrasCaja.fila >= filasMap ||
                posicionDetrasCaja.col < 0 || posicionDetrasCaja.col >= columnasMap) {
                return estado; // Empujar fuera de límites
            }

            // Verificar si el destino del empuje es una pared o está ocupado por otra caja
            if (estado.mapa[posicionDetrasCaja.fila][posicionDetrasCaja.col] === '#') {
                return estado; // Empujar contra pared
            }

            const hayOtraCaja = estado.cajas.some(c => 
                !(c.fila === cajaAEmpujar.fila && c.col === cajaAEmpujar.col) && // No es la caja que empujamos
                c.fila === posicionDetrasCaja.fila && c.col === posicionDetrasCaja.col
            );

            if (hayOtraCaja) {
                return estado; // Empujar contra otra caja
            }

            // ¡Empuje exitoso! Crear el nuevo estado
            nuevoEstado.movimientos += 1;
            nuevoEstado.jugador = nuevaPosicionJugador;

            // Actualizar cajas: la caja se mueve, y todas las demás permanecen iguales
            let nuevasCajas = [];
            estado.cajas.forEach(c => {
                if (c.fila === cajaAEmpujar.fila && c.col === cajaAEmpujar.col) {
                    // Esta es la caja que se mueve
                    nuevasCajas.push({ fila: posicionDetrasCaja.fila, col: posicionDetrasCaja.col });
                } else {
                    // Otras cajas permanecen en su lugar
                    nuevasCajas.push(c);
                }
            });
            nuevoEstado.cajas = nuevasCajas;

            return nuevoEstado;
        }
    },

    /**
     * Verifica si todas las cajas están sobre un objetivo.
     * @param {object} estado - El estado actual del juego.
     * @returns {boolean} True si el nivel está ganado, False en caso contrario.
     */
    ganado: function(estado) {
        if (estado.objetivos.length === 0) return true; // No hay objetivos que cumplir

        const cajasEnObjetivo = estado.cajas.every(caja => {
            // Verificar si la caja está sobre algún objetivo
            return estado.objetivos.some(obj => caja.fila === obj.fila && caja.col === obj.col);
        });

        return cajasEnObjetivo;
    }
};
