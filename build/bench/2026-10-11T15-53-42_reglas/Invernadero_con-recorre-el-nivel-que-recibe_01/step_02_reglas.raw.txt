const Reglas = {
    /**
     * Crea el estado inicial del juego a partir de un nivel dado.
     * @param {string[]} nivel - Array de strings que representan el mapa (filas).
     * @returns {object} El estado inicial del juego.
     */
    crearEstado: function(nivel) {
        const filas = nivel.length;
        if (filas === 0) return null;
        const columnas = nivel[0].length;

        let jugadorPos = null;
        let cajasIniciales = [];
        let objetivosIniciales = [];
        let mapaTemporal = JSON.parse(JSON.stringify(nivel)); // Copia profunda para modificar el mapa base

        // 1. Identificar posiciones iniciales y limpiar el mapa temporal
        for (let r = 0; r < filas; r++) {
            for (let c = 0; c < columnas; c++) {
                const char = nivel[r][c];
                if (char === '@') {
                    jugadorPos = { fila: r, col: c };
                    mapaTemporal[r] = mapaTemporal[r].substring(0, c) + ' ' + mapaTemporal[r].substring(c + 1);
                } else if (char === '$') {
                    cajasIniciales.push({ fila: r, col: c });
                    mapaTemporal[r] = mapaTemporal[r].substring(0, c) + ' ' + mapaTemporal[r].substring(c + 1);
                } else if (char === '.') {
                    objetivosIniciales.push({ fila: r, col: c });
                    // Los objetivos se mantienen en el mapa para la representación visual, pero no son parte de los caracteres base del estado.
                    // Sin embargo, dado que el contrato dice que el mapa solo tiene '#', ' ', '.', y '.' es un objetivo estático, lo dejamos así.
                }
            }
        }

        // 2. Crear el estado final
        const estado = {
            mapa: mapaTemporal, // Mapa con paredes, pisos y objetivos ('.')
            jugador: jugadorPos,
            cajas: cajasIniciales,
            objetivos: objetivosIniciales,
            movimientos: 0
        };

        return estado;
    },

    /**
     * Mueve el estado según la dirección dada. Maneja colisiones y empujes de cajas.
     * @param {object} estado - El estado actual del juego.
     * @param {"arriba" | "abajo" | "izquierda" | "derecha"} direccion - Dirección del movimiento.
     * @returns {object} Un nuevo estado (o el original si no hay movimiento).
     */
    mover: function(estado, direccion) {
        const filas = estado.mapa.length;
        if (filas === 0) return state;
        const columnas = estado.mapa[0].length;

        let nuevoEstado = JSON.parse(JSON.stringify(estado)); // Copia profunda para el nuevo estado
        let { fila: r, col: c } = estado.jugador;
        let nuevaR = r;
        let nuevaC = c;

        // 1. Calcular la posición objetivo del jugador
        switch (direccion) {
            case "arriba":
                nuevaR -= 1;
                break;
            case "abajo":
                nuevaR += 1;
                break;
            case "izquierda":
                nuevaC -= 1;
                break;
            case "derecha":
                nuevaC += 1;
                break;
        }

        // Función auxiliar para verificar si una posición está dentro del mapa y no es pared.
        const isValidPosition = (r, c) => {
            if (r < 0 || r >= filas || c < 0 || c >= columnas) return false; // Fuera de límites
            return estado.mapa[r][c] !== '#'; // No es pared
        };

        // 2. Verificar colisión básica del jugador
        if (!isValidPosition(nuevaR, nuevaC)) {
            return state; // Colisión con pared o límite
        }

        let cajaObjetivo = null;
        let indiceCajaEncontrada = -1;

        // 3. Comprobar si la posición objetivo está ocupada por una caja
        for (let i = 0; i < estado.cajas.length; i++) {
            const box = estado.cajas[i];
            if (box.fila === nuevaR && box.col === nuevaC) {
                cajaObjetivo = box;
                indiceCajaEncontrada = i;
                break;
            }
        }

        // 4. Manejar el movimiento de la caja (si aplica)
        if (cajaObjetivo) {
            let posDetrasR = nuevaR;
            let posDetrasC = nuevaC;

            // Calcular la posición detrás de la caja en la dirección opuesta al movimiento
            switch (direccion) {
                case "arriba":
                    posDetrasR += 1;
                    break;
                case "abajo":
                    posDetrasR -= 1;
                    break;
                case "izquierda":
                    posDetrasC += 1;
                    break;
                case "derecha":
                    posDetrasC -= 1;
                    break;
            }

            // Verificar si el espacio detrás de la caja está libre (no pared y no otra caja)
            const isBehindValid = isValidPosition(posDetrasR, posDetrasC);
            let hayOtraCajaDetras = false;
            for (let i = 0; i < estado.cajas.length; i++) {
                if (i !== indiceCajaEncontrada && estado.cajas[i].fila === posDetrasR && estado.cajas[i].col === posDetrasC) {
                    hayOtraCajaDetras = true;
                    break;
                }
            }

            // Si el espacio detrás está bloqueado (pared o otra caja), no se mueve nada.
            if (!isBehindValid || hayOtraCajaDetras) {
                return state;
            }

            // Movimiento exitoso: Empujar la caja y al jugador
            nuevoEstado.cajas[indiceCajaEncontrada] = { fila: posDetrasR, col: posDetrasC };
            nuevaR = nuevaR; // El jugador se mueve a donde estaba la caja
            nuevaC = nuevaC;

        } else {
            // 5. Movimiento simple del jugador (sin cajas)
            // La posición objetivo es el destino final del jugador.
        }

        // 6. Actualizar estado después de un movimiento exitoso
        nuevoEstado.jugador = { fila: nuevaR, col: nuevaC };
        nuevoEstado.movimientos += 1;

        return nuevoEstado;
    },

    /**
     * Verifica si todas las macetas están sobre sus objetivos.
     * @param {object} estado - El estado actual del juego.
     * @returns {boolean} True si el nivel está ganado, False en caso contrario.
     */
    ganado: function(estado) {
        if (estado.objetivos.length === 0) return true; // No hay objetivos que cubrir

        const cajasEnUbicacion = new Set();
        for (const caja of estado.cajas) {
            cajasEnUbicacion.add(`${caja.fila},${caja.col}`);
        }

        // Comprobar si cada objetivo tiene una caja encima
        for (const objetivo of estado.objetivos) {
            if (!cajasEnUbicacion.has(`${objetivo.fila},${objetivo.col}`)) {
                return false; // Algún objetivo no está cubierto
            }
        }

        return true; // Todos los objetivos están cubiertos
    }
};
