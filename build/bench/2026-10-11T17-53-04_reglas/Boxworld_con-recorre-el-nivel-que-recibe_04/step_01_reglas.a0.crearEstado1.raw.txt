crearEstado: function(nivel) {
        const filas = nivel.length;
        if (filas === 0) return null;
        const columnas = nivel[0].length;

        let jugadorPos = null;
        let cajasArray = [];
        let objetivosArray = [];
        let mapaNuevo = [];

        for (let r = 0; r < filas; r++) {
            let filaString = "";
            for (let c = 0; c < columnas; c++) {
                const char = nivel[r][c];
                let tileChar = '';

                if (char === '@') {
                    jugadorPos = { fila: r, col: c };
                    tileChar = ' '; // El mapa solo guarda elementos estáticos
                } else if (char === '$') {
                    cajasArray.push({ fila: r, col: c });
                    tileChar = ' ';
                } else if (char === '.') {
                    objetivosArray.push({ fila: r, col: c });
                    tileChar = '.'; // Los objetivos se mantienen en el mapa estático
                } else if (char === '#') {
                    tileChar = '#';
                } else {
                    // ' ' (piso)
                    tileChar = ' ';
                }
                filaString += tileChar;
            }
            mapaNuevo.push(filaString);
        }

        return {
            mapa: mapaNuevo, // Mapa con solo #, ' ', '.'
            jugador: jugadorPos,
            cajas: cajasArray,
            objetivos: objetivosArray,
            movimientos: 0
        };
    },
