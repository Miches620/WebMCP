crearEstado: function(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];
        let mapaBase = [];

        // 1. Parsear el nivel para identificar componentes y construir el mapa base
        for (let r = 0; r < nivel.length; r++) {
            let filaString = '';
            for (let c = 0; c < nivel[r].length; c++) {
                const char = nivel[r][c];
                if (char === '@') {
                    jugador = { fila: r, col: c };
                    filaString += ' '; // El jugador no cuenta en el mapa base
                } else if (char === '$') {
                    cajas.push({ fila: r, col: c });
                    filaString += ' '; // La caja no cuenta en el mapa base
                } else if (char === '.') {
                    objetivos.push({ fila: r, col: c });
                    filaString += '.';
                } else if (char === '#') {
                    filaString += '#';
                } else {
                    // Espacio vacío (' ') o piso normal
                    filaString += char;
                }
            }
            mapaBase.push(filaString);
        }

        return {
            mapa: mapaBase,
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    },
