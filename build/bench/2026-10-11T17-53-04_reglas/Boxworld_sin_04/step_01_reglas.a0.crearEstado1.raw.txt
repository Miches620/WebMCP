crearEstado(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];

        const H = nivel.length;
        if (H === 0) return { mapa: [], jugador: null, cajas: [], objetivos: [], movimientos: 0 };

        // Asumimos que todas las filas tienen la misma longitud W
        const W = nivel[0].length; 
        let mapaSinElementos = [];

        for (let r = 0; r < H; r++) {
            let filaActualChars = [];
            for (let c = 0; c < W; c++) {
                const char = nivel[r][c];
                let mapChar = char; // Valor por defecto para el mapa

                if (char === '@') {
                    jugador = { fila: r, col: c };
                    mapChar = ' '; 
                } else if (char === '$') {
                    cajas.push({ fila: r, col: c });
                    mapChar = ' '; 
                } else if (char === '.') {
                    objetivos.push({ fila: r, col: c });
                    // mapChar permanece '.'
                } else if (char !== '#' && char !== ' ') {
                     // Manejo de cualquier carácter inesperado que no debe estar en el mapa final
                     mapChar = ' '; 
                }

                filaActualChars.push(mapChar);
            }
            mapaSinElementos.push(filaActualChars.join(''));
        }

        return {
            mapa: mapaSinElementos,
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    }
