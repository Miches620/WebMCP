crearEstado: function(nivel) {
        const filas = nivel.length;
        if (filas === 0) return null;
        const columnas = nivel[0].length;

        let jugadorPos = null;
        let cajasIniciales = [];
        let objetivosIniciales = [];
        let mapaLimpio = [];

        // Primera pasada: identificar posiciones y construir el mapa base.
        for (let r = 0; r < filas; r++) {
            const filaString = nivel[r];
            let nuevaFila = ''; // Usamos un string para representar la fila completa
            for (let c = 0; c < columnas; c++) {
                const char = filaString[c];
                if (char === '@') {
                    jugadorPos = { fila: r, col: c };
                    nuevaFila += ' '; // El jugador no se cuenta en el mapa base
                } else if (char === '$') {
                    cajasIniciales.push({ fila: r, col: c });
                    nuevaFila += ' '; // La caja no se cuenta en el mapa base
                } else if (char === '.') {
                    objetivosIniciales.push({ fila: r, col: c });
                    nuevaFila += '.';
                } else if (char === '#') {
                    nuevaFila += '#';
                } else { // ' '
                    nuevaFila += ' ';
                }
            }
            mapaLimpio.push(nuevaFila); // Almacenamos la fila como un string
        }

        // Asegurarse de que el jugador y las cajas se inicialicen correctamente si no fueron encontrados (aunque el contrato asegura su presencia).
        if (!jugadorPos) {
             console.error("Error: Jugador '@' no encontrado en el nivel.");
        }


        return {
            mapa: mapaLimpio,
            jugador: jugadorPos,
            cajas: cajasIniciales,
            objetivos: objetivosIniciales,
            movimientos: 0
        };
    },
