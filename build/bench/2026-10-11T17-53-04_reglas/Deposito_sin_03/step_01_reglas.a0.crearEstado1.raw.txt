```javascript
crearEstado: function(nivel) {
    let jugadorPos = null;
    let cajasIniciales = [];
    let objetivosIniciales = [];
    const mapaBase = [];

    // Iteramos sobre las filas (r) y columnas (c) para construir el estado.
    for (let r = 0; r < nivel.length; r++) {
        let rowString = '';
        for (let c = 0; c < nivel[r].length; c++) {
            const char = nivel[r][c];

            // Construir la base del mapa: solo '#', ' ', '.'
            if (char === '#') {
                rowString += '#';
            } else if (char === ' ') {
                rowString += ' ';
            } else if (char === '.') {
                rowString += '.';
                // Si es un objetivo, lo registramos.
                objetivosIniciales.push({ fila: r, col: c });
            } 
            // Ignoramos '$' y '@' para el mapa base

            // Encontrar posiciones especiales
            if (char === '@') {
                jugadorPos = { fila: r, col: c };
            } else if (char === '$') {
                cajasIniciales.push({ fila: r, col: c });
            }
        }
        mapaBase.push(rowString);
    }

    return {
        mapa: mapaBase,
        jugador: jugadorPos,
        cajas: cajasIniciales,
        objetivos: objetivosIniciales,
        movimientos: 0
    };
},
```
