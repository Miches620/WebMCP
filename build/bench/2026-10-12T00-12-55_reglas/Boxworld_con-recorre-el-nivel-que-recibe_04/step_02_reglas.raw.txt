const Reglas = {
    /**
     * Crea el estado inicial del juego a partir de un nivel dado.
     * @param {string[]} nivel - Array de strings que representan el mapa del nivel.
     * @returns {object} El estado inicial del juego.
     */
    crearEstado: function(nivel) {
        const filas = nivel.length;
        const columnas = nivel[0].length;

        let jugadorPos = null;
        let cajasCoords = [];
        let objetivosCoords = [];
        let mapaBase = Array(filas).fill(null).map(() => Array(columnas).fill(''));

        for (let r = 0; r < filas; r++) {
            for (let c = 0; c < columnas; c++) {
                const char = nivel[r][c];
                if (char === '@') {
                    jugadorPos = { fila: r, col: c };
                } else if (char === '$') {
                    cajasCoords.push({ fila: r, col: c });
                } else if (char === '.') {
                    objetivosCoords.push({ fila: r, col: c });
                }

                // Construye el mapa base sin jugador ni cajas
                if (char !== '@' && char !== '$') {
                    mapaBase[r][c] = char;
                } else {
                    mapaBase[r][c] = '#'; // Ocupado por algo que no es parte del piso/objetivo base
                }
            }
        }

        // El mapa final debe ser un array de strings, como el input.
        const mapaStringArray = mapaBase.map(row => row.join(''));

        return {
            mapa: mapaStringArray,
            jugador: jugadorPos,
            cajas: cajasCoords,
            objetivos: objetivosCoords,
            movimientos: 0
        };
    },

    /**
     * Mueve el estado del juego según la dirección y las reglas de colisión.
     * @param {object} estado - El estado actual del juego.
     * @param {"arriba" | "abajo" | "izquierda" | "derecha"} direccion - Dirección del movimiento.
     * @returns {object} Un nuevo estado (inmutable) con el movimiento aplicado o sin cambios si es inválido.
     */
    mover: function(estado, direccion) {
        const R = estado.mapa.length;
        const C = estado.mapa[0].length;

        // 1. Calcular la posición objetivo del jugador (sin considerar cajas aún)
        let nextPlayerPos = { ...estado.jugador };
        switch (direccion) {
            case "arriba":
                nextPlayerPos.fila -= 1;
                break;
            case "abajo":
                nextPlayerPos.fila += 1;
                break;
            case "izquierda":
                nextPlayerPos.col -= 1;
                break;
            case "derecha":
                nextPlayerPos.col += 1;
                break;
        }

        // Verificar límites del mapa
        if (nextPlayerPos.fila < 0 || nextPlayerPos.fila >= R || nextPlayerPos.col < 0 || nextPlayerPos.col >= C) {
            return { ...estado, movimientos: estado.movimientos }; // Movimiento fuera de límites
        }

        // Verificar si el destino es una pared (usando el mapa base)
        if (estado.mapa[nextPlayerPos.fila][nextPlayerPos.col] === '#') {
            return { ...estado, movimientos: estado.movimientos }; // Colisión con pared
        }

        // 2. Verificar si hay una caja en la posición objetivo
        const boxAtTarget = estado.cajas.some(box => 
            box.fila === nextPlayerPos.fila && box.col === nextPlayerPos.col
        );

        if (!boxAtTarget) {
            // Movimiento simple (a piso o objetivo)
            return {
                mapa: [...estado.mapa], // Copia superficial del mapa
                jugador: nextPlayerPos,
                cajas: estado.cajas.map(c => ({ ...c })), // Clonar cajas
                objetivos: state.objetivos.map(o => ({ ...o })),
                movimientos: estado.movimientos + 1
            };
        } else {
            // Hay una caja, intentar empujar
            const boxToPush = estado.cajas.find(box => 
                box.fila === nextPlayerPos.fila && box.col === nextPlayerPos.col
            );

            // Calcular la posición donde debe ir la caja después de ser empujada
            let pushTargetPos = { ...boxToPush };
            switch (direccion) {
                case "arriba":
                    pushTargetPos.fila -= 1;
                    break;
                case "abajo":
                    pushTargetPos.fila += 1;
                    break;
                case "izquierda":
                    pushTargetPos.col -= 1;
                    break;
                case "derecha":
                    pushTargetPos.col += 1;
                    break;
            }

            // Verificar si el destino del empuje es válido:
            // 1. Debe estar dentro de límites.
            if (pushTargetPos.fila < 0 || pushTargetPos.fila >= R || pushTargetPos.col < 0 || pushTargetPos.col >= C) {
                return { ...estado, movimientos: estado.movimientos }; // Colisión con borde
            }

            // 2. El destino del empuje no puede ser una pared (usando el mapa base).
            if (estado.mapa[pushTargetPos.fila][pushTargetPos.col] === '#') {
                return { ...estado, movimientos: estado.movimientos }; // Colisión con pared al empujar
            }

            // 3. El destino del empuje no puede estar ocupado por otra caja (excepto la que se mueve).
            const collisionWithOtherBox = estado.cajas.some(box => 
                box !== boxToPush && box.fila === pushTargetPos.fila && box.col === pushTargetPos.col
            );

            if (collisionWithOtherBox) {
                return { ...estado, movimientos: estado.movimientos }; // Colisión con otra caja
            }

            // ¡Empuje exitoso! Crear el nuevo estado.
            const newCajas = estado.cajas.map(box => {
                if (box === boxToPush) {
                    return pushTargetPos; // La caja se mueve a la nueva posición
                }
                return box; // Otras cajas permanecen igual
            });

            // El jugador ocupa el lugar de la caja original
            const newPlayer = nextPlayerPos; 

            return {
                mapa: [...estado.mapa],
                jugador: newPlayer,
                cajas: newCajas,
                objetivos: estado.objetivos.map(o => ({ ...o })),
                movimientos: estado.movimientos + 1
            };
        }
    },

    /**
     * Verifica si todas las cajas están sobre objetivos.
     * @param {object} estado - El estado actual del juego.
     * @returns {boolean} True si el nivel está ganado, False en caso contrario.
     */
    ganado: function(estado) {
        if (estado.objetivos.length === 0) return true; // Caso borde sin objetivos

        const cajasEnObjetivo = estado.cajas.every(box => 
            estado.objetivos.some(obj => obj.fila === box.fila && obj.col === box.col)
        );

        return cajasEnObjetivo;
    }
};
