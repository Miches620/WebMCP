const Reglas = {
    /**
     * Crea el estado inicial del juego a partir de un nivel dado.
     * @param {string[]} nivel - Array de strings que representan el mapa del nivel.
     * @returns {object} El estado inicial del juego.
     */
    crearEstado: function(nivel) {
        const filas = nivel.length;
        if (filas === 0) return null;
        const columnas = nivel[0].length;

        let jugadorPos = null;
        let cajasIniciales = [];
        let objetivosIniciales = [];
        let mapaNuevo = [];

        // 1. Parsear el nivel para obtener posiciones y construir el mapa base
        for (let r = 0; r < filas; r++) {
            let filaString = nivel[r];
            let nuevaFilaMapa = '';
            for (let c = 0; c < columnas; c++) {
                const char = filaString[c];

                if (char === '@') {
                    jugadorPos = { fila: r, col: c };
                    nuevaFilaMapa += ' '; // El jugador no se dibuja en el mapa base
                } else if (char === '$') {
                    cajasIniciales.push({ fila: r, col: c });
                    nuevaFilaMapa += ' '; // La caja no se dibuja en el mapa base
                } else if (char === '.') {
                    objetivosIniciales.push({ fila: r, col: c });
                    nuevaFilaMapa += '.';
                } else if (char === '#') {
                    nuevaFilaMapa += '#';
                } else { // ' '
                    nuevaFilaMapa += ' ';
                }
            }
            mapaNuevo.push(nuevaFilaMapa);
        }

        // 2. Crear el estado inicial
        return {
            mapa: mapaNuevo,
            jugador: jugadorPos,
            cajas: cajasIniciales,
            objetivos: objetivosIniciales,
            movimientos: 0
        };
    },

    /**
     * Calcula un nuevo estado después de intentar mover al jugador.
     * @param {object} estado - El estado actual del juego.
     * @param {"arriba" | "abajo" | "izquierda" | "derecha"} direccion - Dirección deseada.
     * @returns {object} Un nuevo estado o el estado original si no hay movimiento posible.
     */
    mover: function(estado, direccion) {
        const jugador = estado.jugador;
        let nuevaFilaJugador = jugador.fila;
        let nuevaColJugador = jugador.col;

        // 1. Calcular la posición objetivo del jugador
        switch (direccion) {
            case "arriba":
                nuevaFilaJugador -= 1;
                break;
            case "abajo":
                nuevaFilaJugador += 1;
                break;
            case "izquierda":
                nuevaColJugador -= 1;
                break;
            case "derecha":
                nuevaColJugador += 1;
                break;
        }

        // Verificar límites del mapa (8x8)
        const filas = estado.mapa.length;
        const columnas = estado.mapa[0].length;

        if (nuevaFilaJugador < 0 || nuevaFilaJugador >= filas || nuevaColJugador < 0 || nuevaColJugador >= columnas) {
            return { ...estado, movimientos: estado.movimientos }; // Fuera de límites
        }

        // Verificar si el destino es una pared (el mapa base solo tiene #, ' ', '.')
        if (estado.mapa[nuevaFilaJugador][nuevaColJugador] === '#') {
            return { ...estado, movimientos: estado.movimientos }; // Colisión con pared
        }

        // 2. Verificar colisiones con cajas
        const cajaEnDestino = estado.cajas.find(c => c.fila === nuevaFilaJugador && c.col === nuevaColJugador);

        if (!cajaEnDestino) {
            // Caso A: Movimiento libre (no hay caja en el destino)
            return this._crearEstadoMovimientoLibre(estado, nuevaFilaJugador, nuevaColJugador);
        } else {
            // Caso B: Colisión con una caja. Intentar empujar.
            const box = cajaEnDestino;
            let filaEmpuje = box.fila;
            let colEmpuje = box.col;

            // Calcular la posición detrás de la caja (destino del empuje)
            switch (direccion) {
                case "arriba":
                    filaEmpuje -= 1;
                    break;
                case "abajo":
                    filaEmpuje += 1;
                    break;
                case "izquierda":
                    colEmpuje -= 1;
                    break;
                case "derecha":
                    colEmpuje += 1;
                    break;
            }

            // Verificar si el destino del empuje es válido (no pared, no fuera de límites)
            if (filaEmpuje < 0 || filaEmpuje >= filas || colEmpuje < 0 || colEmpuje >= columnas || estado.mapa[filaEmpuje][colEmpuje] === '#') {
                return { ...estado, movimientos: estado.movimientos }; // Empuje bloqueado por pared o límite
            }

            // Verificar si el destino del empuje está ocupado por otra caja
            const hayOtraCajaDetras = estado.cajas.some(c => c !== box && c.fila === filaEmpuje && c.col === colEmpuje);

            if (hayOtraCajaDetras) {
                return { ...estado, movimientos: estado.movimientos }; // Empuje bloqueado por otra caja
            }

            // Éxito en el empuje
            const nuevoEstado = this._crearEstadoMovimientoEmpujando(estado, box, filaEmpuje, colEmpuje);
            return nuevoEstado;
        }
    },

    /**
     * Maneja el movimiento del jugador sin cajas involucradas.
     * @private
     */
    _crearEstadoMovimientoLibre: function(estado, nuevaFilaJugador, nuevaColJugador) {
        // Crear un nuevo estado inmutable
        const nuevoEstado = {
            mapa: [...estado.mapa], // Copia superficial del mapa (solo strings)
            jugador: { fila: nuevaFilaJugador, col: nuevaColJugador },
            cajas: [...estado.cajas],
            objetivos: estado.objetivos,
            movimientos: estado.movimientos + 1
        };
        return nuevoEstado;
    },

    /**
     * Maneja el movimiento del jugador y empuja una caja exitosamente.
     * @private
     */
    _crearEstadoMovimientoEmpujando: function(estado, boxOriginal, nuevaFilaBox, nuevaColBox) {
        // 1. Actualizar la posición de las cajas (copiar el array e insertar cambios)
        const nuevasCajas = estado.cajas.map(c => ({ ...c })); // Clonar todas las cajas

        // Encontrar y actualizar la caja que se movió
        const indexBox = nuevasCajas.findIndex(c => c.fila === boxOriginal.fila && c.col === boxOriginal.col);
        if (indexBox !== -1) {
            nuevasCajas[indexBox] = { fila: nuevaFilaBox, col: nuevaColBox };
        }

        // 2. Crear el nuevo estado
        const nuevoEstado = {
            mapa: [...estado.mapa],
            jugador: { fila: boxOriginal.fila, col: boxOriginal.col }, // El jugador ocupa la posición anterior de la caja
            cajas: nuevasCajas,
            objetivos: estado.objetivos,
            movimientos: estado.movimientos + 1
        };

        return nuevoEstado;
    },


    /**
     * Verifica si todas las cajas están sobre un objetivo.
     * @param {object} estado - El estado actual del juego.
     * @returns {boolean} True si el nivel está ganado, False en caso contrario.
     */
    ganado: function(estado) {
        if (estado.objetivos.length === 0) return true; // No hay objetivos que cubrir

        const cajasEnObjetivo = estado.cajas.filter(c =>
            estado.objetivos.some(o => o.fila === c.fila && o.col === c.col)
        );

        // Comprobar si el número de cajas en objetivo es igual al total de objetivos
        if (cajasEnObjetivo.length !== estado.objetivos.length) {
            return false;
        }

        // Verificar que cada objetivo esté cubierto por una caja única
        const ubicacionesCaja = new Set(estado.cajas.map(c => `${c.fila},${c.col}`));
        const ubicacionesObjetivo = new Set(estado.objetivos.map(o => `${o.fila},${o.col}`));

        for (const objLoc of ubicacionesObjetivo) {
            if (!ubicacionesCaja.has(objLoc)) {
                return false; // Algún objetivo no tiene caja encima
            }
        }

        // Si todos los objetivos tienen una caja y el número de cajas es suficiente, se ha ganado.
        return true;
    }
};
