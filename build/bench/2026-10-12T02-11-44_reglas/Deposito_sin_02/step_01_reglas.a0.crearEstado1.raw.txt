crearEstado: function(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];
        let nuevoMapa = [];

        for (let r = 0; r < 8; r++) {
            // Safety check: ensure the row string exists, defaulting to empty string if undefined/null.
            const rowString = nivel[r] || '';
            nuevoMapa.push([]);
            
            for (let c = 0; c < 8; c++) {
                // Get character safely. If the input row is shorter than 8, char will be undefined.
                let char = rowString[c];
                
                // Determine the effective character for processing. Default to space if out of bounds.
                const effectiveChar = (char === undefined) ? ' ' : char;

                if (effectiveChar === '@') {
                    jugador = { fila: r, col: c };
                    nuevoMapa[r].push(' '); // El jugador no se dibuja en el mapa base
                } else if (effectiveChar === '$') {
                    cajas.push({ fila: r, col: c });
                    nuevoMapa[r].push(' '); // La caja no se dibuja en el mapa base
                } else if (effectiveChar === '.') {
                    objetivos.push({ fila: r, col: c });
                    nuevoMapa[r].push('.');
                } else if (effectiveChar === '#') {
                    nuevoMapa[r].push('#');
                } else { // ' ' o cualquier otro carácter que no sea especial
                    nuevoMapa[r].push(' ');
                }
            }
        }

        // Convertir el array de arrays a un array de strings para mantener la estructura del contrato.
        const mapaString = nuevoMapa.map(row => row.join(''));

        return {
            mapa: mapaString,
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    },
