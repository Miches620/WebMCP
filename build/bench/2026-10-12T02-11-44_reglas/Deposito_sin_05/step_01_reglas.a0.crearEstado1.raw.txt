```javascript
crearEstado: function(nivel) {
    let jugador = null;
    let cajas = [];
    let objetivos = [];

    const numRows = nivel.length;
    if (numRows === 0) return { mapa: [], jugador: null, cajas: [], objetivos: [], movimientos: 0 };

    // Determinar el número de columnas basándose en la primera fila
    const numCols = nivel[0] ? nivel[0].length : 0;

    // Recorrer el nivel para identificar posiciones clave
    for (let r = 0; r < numRows; r++) {
        const rowString = String(nivel[r]); // Asegurar que es un string iterable
        if (!rowString) continue;

        for (let c = 0; c < numCols; c++) {
            const char = rowString[c];
            if (char === '@') {
                jugador = { fila: r, col: c };
            } else if (char === '$') {
                cajas.push({ fila: r, col: c });
            } else if (char === '.') {
                objetivos.push({ fila: r, col: c });
            }
        }
    }

    // El mapa de estado solo debe contener '#', ' ', '.'
    const mapaLimpiado = nivel.map(fila => String(fila).replace(/[@\$]/g, ''));

    return {
        mapa: mapaLimpiado,
        jugador: jugador,
        cajas: cajas,
        objetivos: objetivos,
        movimientos: 0
    };
}
```
