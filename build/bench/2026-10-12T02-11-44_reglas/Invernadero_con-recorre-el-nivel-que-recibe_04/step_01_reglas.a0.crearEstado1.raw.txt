crearEstado: function(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];

        // 1. Recorrer el nivel para identificar elementos y construir las coordenadas iniciales
        for (let r = 0; r < nivel.length; r++) {
            const filaActual = nivel[r];
            for (let c = 0; c < filaActual.length; c++) {
                const char = nivel[r][c];
                if (char === '@') {
                    jugador = { fila: r, col: c };
                } else if (char === '$') {
                    cajas.push({ fila: r, col: c });
                } else if (char === '.') {
                    objetivos.push({ fila: r, col: c });
                }
            }
        }

        // 2. Limpiar el mapa para que solo contenga '#', ' ', '.'
        let mapaFinal = [];
        for (let r = 0; r < nivel.length; r++) {
            const filaStrings = [];
            for (let c = 0; c < nivel[r].length; c++) {
                const char = nivel[r][c];
                if (char === '#') {
                    filaStrings.push('#');
                } else if (char === '.') {
                    filaStrings.push('.');
                } else { // ' ', '@', '$' se convierten en espacio o son ignorados si no es objetivo
                    filaStrings.push(' ');
                }
            }
            mapaFinal.push(filaStrings);
        }

        return {
            mapa: mapaFinal,
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    },
