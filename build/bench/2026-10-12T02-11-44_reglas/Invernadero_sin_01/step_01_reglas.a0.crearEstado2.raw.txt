crearEstado(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];
        const nuevoMapa = [];

        for (let r = 0; r < nivel.length; r++) {
            let nuevaFilaChars = [];
            for (let c = 0; c < nivel[r].length; c++) {
                const char = nivel[r][c];
                let mapChar = char;

                if (char === '@') {
                    jugador = { fila: r, col: c };
                    mapChar = ' '; // El jugador no se dibuja en el mapa de estado
                } else if (char === '$') {
                    cajas.push({ fila: r, col: c });
                    mapChar = ' '; // La caja no se dibuja en el mapa de estado
                } else if (char === '.') {
                    objetivos.push({ fila: r, col: c });
                    // mapChar permanece '.'
                }

                nuevaFilaChars.push(mapChar);
            }
            nuevoMapa.push(nuevaFilaChars.join(''));
        }

        return {
            mapa: nuevoMapa,
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    }
