crearEstado: function(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];

        const numRows = nivel.length;
        if (numRows === 0) {
            return { mapa: [], jugador: null, cajas: [], objetivos: [], movimientos: 0 };
        }
        // Asumimos que el nivel es rectangular y no está vacío
        const numCols = nivel[0].length;

        for (let r = 0; r < numRows; r++) {
            if (!nivel[r]) continue; // Saltar si la fila no existe o es nula
            
            for (let c = 0; c < numCols; c++) {
                const char = nivel[r][c];
                if (char === '@') {
                    jugador = { fila: r, col: c };
                } else if (char === '$') {
                    cajas.push({ fila: r, col: c });
                } else if (char === '.') {
                    objetivos.push({ fila: r, col: c });
                }
            }
        }

        // El mapa solo contiene paredes, pisos y objetivos estáticos.
        const mapa = nivel; 

        return {
            mapa: mapa,
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    },
