crearEstado: function(nivel) {
        let jugador = null;
        let cajas = [];
        let objetivos = [];

        // Inicializar el mapa limpio (solo #, ' ', '.') y encontrar posiciones clave
        const mapaLimpio = nivel.map(fila => 
            Array.from(fila).filter(char => char !== '@' && char !== '$').join('')
        );

        for (let r = 0; r < nivel.length; r++) {
            const fila = nivel[r];
            for (let c = 0; c < fila.length; c++) {
                const char = fila[c];
                if (char === '@') {
                    jugador = { fila: r, col: c };
                } else if (char === '$') {
                    cajas.push({ fila: r, col: c });
                } else if (char === '.') {
                    objetivos.push({ fila: r, col: c });
                }
            }
        }

        return {
            mapa: mapaLimpio, // Mapa sin jugadores ni cajas
            jugador: jugador,
            cajas: cajas,
            objetivos: objetivos,
            movimientos: 0
        };
    },
