```css
/* ========================================= */
/* 11. SECCIÓN CONTACTO (R3)                */
/* ========================================= */

#contacto {
    background-color: var(--bg-light); /* Fondo suave para la sección de contacto */
}

/* Contenedor principal del layout de contacto (Grid) */
.contact-grid {
    display: grid;
    grid-template-columns: 2fr 1fr; /* Formulario más ancho que info */
    gap: var(--spacing-lg);
    align-items: flex-start; /* Alinea los elementos al inicio */
}

/* Estilo general del formulario de contacto */
.contact-form {
    background-color: #ffffff;
    padding: 30px;
    border-radius: var(--border-radius);
    box-shadow: 0 4px 20px rgba(0, 0, 0, 0.08);
}

/* Estilo de los grupos de formulario (Label + Input) */
.form-group {
    margin-bottom: var(--spacing-md);
}

/* Etiqueta del campo (label) */
.form-group label {
    display: block; /* Hace que la etiqueta ocupe su propia línea */
    font-weight: 600;
    color: var(--text-dark);
    margin-bottom: 8px;
}

/* Estilo de los inputs y textarea */
.form-group input[type="text"],
.form-group input[type="email"],
.form-group textarea {
    width: 100%;
    padding: 12px;
    border: 1px solid #ced4da; /* Borde gris estándar */
    border-radius: var(--border-radius);
    font-size: 1rem;
    transition: border-color 0.3s ease, box-shadow 0.3s ease;
}

/* Efecto focus para inputs (mejor UX) */
.form-group input:focus,
.form-group textarea:focus {
    border-color: var(--primary-color); /* Borde azul al enfocar */
    box-shadow: 0 0 0 3px rgba(0, 123, 255, 0.15); /* Sombra suave de acento */
    outline: none; /* Quitar el borde por defecto del navegador */
}

/* Estilo específico para el botón de envío dentro del formulario */
.submit-form-btn {
    margin-top: var(--spacing-md);
    width: 100%; /* Ocupar todo el ancho disponible */
    padding: 1rem;
}


/* Contenedor de información de contacto (columna derecha) */
.contact-info-container {
    background-color: #ffffff;
    padding: 30px;
    border-radius: var(--border-radius);
    box-shadow: 0 4px 20px rgba(0, 0, 0, 0.08);
}

.contact-info-container h3 {
    color: var(--primary-color);

    padding-bottom: 10px;
}

/* Estilo de cada ítem de información (Teléfono, Email, etc.) */

    margin-bottom: var(--spacing-md);
}

.info-item h4 {
    font-size: 1rem;

}
```
