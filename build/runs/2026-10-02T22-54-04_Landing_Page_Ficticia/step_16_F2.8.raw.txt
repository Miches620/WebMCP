/* ========================================= */
/* 12. FOOTER (R4)                           */
/* ========================================= */

#site-footer {
    background-color: var(--bg-dark); /* Fondo oscuro para contraste */
    color: #ccc; /* Texto gris claro sobre fondo oscuro */
    padding: 3rem 0 1.5rem; /* Padding generoso, más espacio abajo */
    border-top: 4px solid var(--primary-color); /* Línea de acento superior */
}

/* Contenedor principal del footer */
.footer-content {
    display: flex;
    flex-wrap: wrap; /* Permite que los elementos se envuelvan en móvil */
    justify-content: space-between; /* Distribuye el espacio entre columnas */
    gap: 30px; /* Espacio general entre las secciones del footer */
    padding-bottom: 20px;
}

/* Estilo de la columna de información (Logo/Contacto) */
.footer-col {
    min-width: 200px; /* Asegura un ancho mínimo para el contenido */
    flex: 1; /* Permite que crezca y se distribuya bien */
    max-width: 30%; /* Limita su crecimiento en escritorio */
}

/* Estilo del logo/nombre dentro del footer */
.footer-col h4 {
    font-size: 1.5rem;
    color: var(--text-light);
    margin-bottom: 1rem;
}

/* Título de las secciones de enlaces (ej. Servicios, Legal) */
.footer-col h5 {
    font-size: 1.1rem;
    color: var(--primary-color);
    margin-bottom: 1rem;
    text-transform: uppercase;
}

/* Lista de enlaces dentro del footer (ul/li o nav) */
.footer-col ul {
    list-style: none;
    display: flex;
    flex-direction: column; /* Apilados verticalmente */
    gap: 8px;
}

/* Estilo de los enlaces específicos del footer */
.footer-col a {
    color: #ccc;
    text-decoration: none;
    font-size: 0.95rem;
    transition: color 0.3s ease, background-color 0.3s ease;
}

/* Efecto hover para enlaces del footer */
.footer-col a:hover {
    color: var(--primary-color); /* Color de acento al pasar el ratón */
    text-decoration: underline; /* Subrayado visible en hover */
    opacity: 1; /* Asegura visibilidad */
}

/* Estilo del copyright y la línea final */
.footer-bottom {
    width: 100%; /* Ocupa todo el ancho disponible */
    text-align: center;
    padding-top: 20px;
    border-top: 1px solid #444; /* Separador sutil */
}

.footer-bottom p {
    margin: 0;
    font-size: 0.9rem;
    color: #888; /* Gris más apagado para el copyright */
}


/* ========================================= */
/* RESPONSIVE PARA FOOTER (Móvil)           */
/* ========================================= */

@media (max-width: 768px) {
    .footer-content {
        flex-direction: column; /* Apilar todas las columnas en móvil */
        align-items: center;
        text-align: center;
        gap: 20px;
    }

    /* Asegurar que todos los elementos ocupen el ancho completo o casi completo */
    .footer-col {
        max-width: 100%;
        text-align: left; /* Mantener la alineación de texto a la izquierda dentro del bloque */
    }

    /* Ajustar el logo en móvil para centrarlo mejor si es necesario */
    .footer-col h4 {
        text-align: center;
    }
}
