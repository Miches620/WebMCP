/* ENHANCEMENT: HOVER AVANZADO PARA TARJETAS   */

/* Refinamiento del efecto hover para las tarjetas de características (Feature Cards) */
/* Añadimos un cambio sutil en el fondo y una sombra más definida con un toque de color primario. */
.feature-card:hover {
    transform: translateY(-8px); /* Mayor elevación al pasar el ratón */
    box-shadow: 0 15px 35px rgba(0, 0, 0, 0.1) translate(0, -2px), 0 0 15px var(--primary-color); /* Sombra más profunda y un toque de color primario (glow) */
    background-color: #f9f9ff; /* Fondo ligeramente diferente para indicar interacción */
}

/* Aseguramos que el icono también reaccione sutilmente al hover del contenedor padre */
.feature-card:hover .feature-icon {
    transform: scale(1.05);
    transition: transform 0.3s ease;
}
