### FILE: site-footer.js
```javascript
/**
 * Inicializa las animaciones de aparición al cargar el footer.
 * Se utiliza IntersectionObserver para simular la animación de scroll (R5).
 */
function animateFooterElements() {
    const observerOptions = {
        root: null,
        rootMargin: '0px',
        threshold: 0.1 // Dispara cuando el 10% del elemento es visible
    };

    // Selecciona todos los elementos que queremos animar (ej. las columnas)
    const elementsToObserve = document.querySelectorAll('.footer-col');

    const observer = new IntersectionObserver((entries, observer) => {
        entries.forEach(entry => {
            if (entry.isIntersecting) {
                // Añadir una clase para activar la animación CSS
                entry.target.classList.add('visible');
                // Dejar de observar el elemento después de animarlo
                observer.unobserve(entry.target); 
            }
        });
    }, observerOptions);

    elementsToObserve.forEach(element => {
        // Inicialmente, añadimos una clase para que CSS pueda aplicar la animación de 'oculto'
        element.style.opacity = '0';
        element.style.transform = 'translateY(20px)';
        observer.observe(element);
    });
}

// Ejecutar la función al cargar el DOM
document.addEventListener('DOMContentLoaded', animateFooterElements);
```
