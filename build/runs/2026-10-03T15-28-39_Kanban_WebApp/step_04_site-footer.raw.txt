#site-footer {
    background-color: var(--color-bg-primary); /* Usando el fondo principal */
    padding: calc(var(--spacing-lg) * 2) 0;
    border-top: 1px solid var(--color-border);
}

.footer-content {
    display: flex;
    flex-wrap: wrap;
    justify-content: space-between;
    align-items: flex-start; /* Alinea los elementos al inicio */
    gap: var(--spacing-lg);
}

.footer-section {
    min-width: 150px; /* Asegura un ancho mínimo para las columnas */
    flex: 1 1 calc(25% - var(--spacing-md)); /* Permite que crezcan y se encojan, ocupando aproximadamente 4 columnas */
}

/* Ajuste específico para el copyright para que no sea tan grande */
.copyright-info {
    flex: 1 1 100%; /* Ocupa todo el ancho en la primera fila */
    text-align: right;
    margin-bottom: var(--spacing-md);
}

/* Estilos de títulos y listas */
.footer-section h4, .links-group h4 {
    color: var(--color-text-secondary);
    font-size: 1rem;
    margin-bottom: var(--spacing-sm);
    text-transform: uppercase;
}

.footer-section ul {
    list-style: none;
    padding: 0;
    margin: 0;
}

.footer-section li {
    margin-bottom: var(--spacing-xs);
}

.footer-section a {
    color: var(--color-text-secondary);
    font-size: 0.9rem;
    transition: color var(--transition-speed);
}

.footer-section a:hover {
    color: var(--color-accent-primary); /* Usamos el acento principal al pasar el mouse */
    opacity: 1;
}

/* Estilos de redes sociales */
.social-icons a {
    display: inline-block;
    margin-right: var(--spacing-sm);
    font-size: 1.5rem; /* Tamaño visible para los emojis/iconos */
    color: var(--color-text-secondary);
}

.social-icons a:hover {
    transform: scale(1.1);
    opacity: 0.8;
}


/* Media Query para dispositivos móviles (390px) */
@media (max-width: 600px) {
    .footer-content {
        flex-direction: column; /* Apila los elementos verticalmente */
        align-items: center;
        text-align: center;
    }

    .footer-section {
        min-width: 100%; /* Ocupa todo el ancho en móvil */
        margin-bottom: var(--spacing-md);
        padding: var(--spacing-sm) 0;
        border-bottom: 1px dashed var(--color-border);
    }

    .footer-section:last-child {
        border-bottom: none; /* Quitar borde al último elemento */
    }

    /* Ajustar el copyright para que esté centrado en móvil si se apila */
    .copyright-info {
        text-align: center !important;
        margin-top: var(--spacing-md);
        border-bottom: none;
    }
}
