    #site-footer .container {

    #site-footer .footer-links,
    #site-footer .footer-copyright {
        min-width: 100%;
        margin-bottom: 2rem !important;
        text-align: center;
    }

    /* Ajusta la disposición de los enlaces para que se apilen verticalmente */
    #site-footer .footer-links {
        flex-direction: column;
        gap: 1.5rem;
    }
    
    /* Hace que los títulos sean bloques completos en móvil */
    #site-footer h4 {
        display: block;
        text-align: center;
    }
}
