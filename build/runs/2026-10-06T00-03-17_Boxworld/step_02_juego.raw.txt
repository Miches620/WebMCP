const estado = {
    levelIndex: 0,
    maxLevel: 5,
    mapWidth: 10,
    mapHeight: 10,
    playerPos: { r: 0, c: 0 },
    boxes: [], // Array de posiciones actuales de las cajas [{r, c}]

    moves: 0,
    isGameOver: false,
};

// =============================================================================
// DATOS DE LOS NIVELES (R3)

const niveles = [

// =============================================================================
// FUNCIONES DE UTILIDAD Y LÓGICA DEL JUEGO
// =============================================================================

/**
 * Inicializa el estado del juego para un nuevo nivel.
 */
function resetEstadoNivel(levelIndex) {
    const levelData = niveles[levelIndex];
    if (!levelData) return false;

    estado.playerPos = { ...levelData.playerStart };
    // Clonar las posiciones iniciales de las cajas para evitar mutaciones directas
    estado.boxes = JSON.parse(JSON.stringify(levelData.boxesInitial)); 
    estado.walls = levelData.walls;
    estado.objectives = levelData.objectives;
    estado.moves = 0;
    estado.isGameOver = false;

    return true;
}

/**
 * Verifica si una coordenada (r, c) está dentro de los límites del mapa y no es una pared.
 * @param {number} r - Fila (row).
 * @param {number} c - Columna (col).
 * @returns {boolean} True si la posición es válida para moverse.
 */
function isValidMove(r, c) {
    if (r < 0 || r >= estado.mapHeight || c < 0 || c >= estado.mapWidth) {
        return false; // Fuera de límites
    }
    // Chequear si es una pared fija
    const isWall = estado.walls.some(wall => wall.r === r && wall.c === c);
    if (isWall) return false;

    // Chequear si ya está ocupado por otra caja (el jugador puede estar sobre un objetivo, pero no debe colisionar con una caja en movimiento).
    const isBoxOccupied = estado.boxes.some(box => box.r === r && box.c === c);
    return true;
}

/**
 * Intenta mover al avatar a la nueva posición (nextR, nextC).
 * Implementa la lógica de colisión y empuje.
 * @param {number} direction - Dirección del movimiento: 0=Arriba, 1=Derecha, 2=Abajo, 3=Izquierda.
 * @returns {boolean} True si el movimiento fue exitoso (y se actualizó el estado).
 */
function attemptMove(direction) {
    if (estado.isGameOver) return false;

    let nextR = estado.playerPos.r;
    let nextC = estado.playerPos.c;

    // Calcular la posición deseada del jugador
    switch (direction) {
        case 0: nextR--; break; // Arriba
        case 1: nextC++; break; // Derecha
        case 2: nextR++; break; // Abajo
        case 3: nextC--; break; // Izquierda
    }

    // 1. Validación de movimiento básico (paredes/límites)
    if (!isValidMove(nextR, nextC)) {
        return false;
    }

    // 2. Chequear colisión con cajas
    const collidingBoxIndex = estado.boxes.findIndex(box => box.r === nextR && box.c === nextC);

    if (collidingBoxIndex !== -1) {
        // Colisión detectada: Intentar empujar la caja
        const boxToPush = estado.boxes[collidingBoxIndex];
        let pushR = boxToPush.r;
        let pushC = boxToPush.c;

        // Calcular la posición donde debe ir la caja (detrás del jugador)
        switch (direction) {
            case 0: pushR--; break; // Arriba
            case 1: pushC++; break; // Derecha
            case 2: pushR++; break; // Abajo
            case 3: pushC--; break; // Izquierda
        }

        // Validación de empuje: ¿Es el destino del box válido? (No pared, no fuera de límites)
        if (isValidMove(pushR, pushC)) {
            // Chequear si la posición objetivo está ocupada por otra caja.
            const targetOccupiedByOtherBox = estado.boxes.some((box, index) => 
                index !== collidingBoxIndex && box.r === pushR && box.c === pushC
            );

            if (!targetOccupiedByOtherBox) {
                // ¡Empuje exitoso! Actualizar el estado de la caja y del jugador.
                estado.boxes[collidingBoxIndex] = { r: pushR, c: pushC };
                estado.playerPos = { r: nextR, c: nextC };
                estado.moves++; // Incremento de movimiento (F3.5)
                return true;
            }
        }
        // Si el empuje falla (pared o caja), no se mueve nada.
        return false;

    } else {
        // 3. Movimiento libre exitoso (no colisiona con cajas ni paredes)
        estado.playerPos = { r: nextR, c: nextC };
        estado.moves++; // Incremento de movimiento (F3.5)
        return true;
    }
}

/**
 * Verifica si todas las cajas están en sus posiciones objetivo.
 * @returns {boolean} True si el nivel se ha superado.
 */
function checkWinCondition() {
    if (estado.boxes.length !== estado.objectives.length) return false;

    let allMatched = true;
    for (let i = 0; i < estado.boxes.length; i++) {
        const boxPos = estado.boxes[i];
        const objectivePos = estado.objectives[i];

        if (boxPos.r !== objectivePos.r || boxPos.c !== objectivePos.c) {
            allMatched = false;
            break;
        }
    }
    return allMatched;
}


// =============================================================================
// RENDERIZADO Y MANEJO DE EVENTOS (A3, A4)
// =============================================================================

/**
 * Renderiza el estado actual del juego en el DOM.
 */
function dibujar() {
    const mapaEl = document.getElementById('mapa');
    if (!mapaEl) return;

    // Limpiar y reconstruir la grilla (F1.6)
    mapaEl.innerHTML = ''; 
    
    // Crear todos los casilleros de 10x10
    for (let r = 0; r < estado.mapHeight; r++) {
        for (let c = 0; c < estado.mapWidth; c++) {
            const cell = document.createElement('div');
            cell.className = 'cell';
            cell.setAttribute('data-row', r);
            cell.setAttribute('data-col', c);

            // Determinar el tipo de casillero (Pared, Objetivo)
            let type = '';
            if (estado.walls.some(w => w.r === r && w.c === c)) {
                type = 'pared';
            } else if (estado.objectives.some(o => o.r === r && o.c === c)) {
                type = 'objetivo';
            }

            cell.setAttribute('data-type', type);
            mapaEl.appendChild(cell);
        }
    }

    // 1. Renderizar elementos dinámicos (Cajas, Avatar)
    const cells = mapaEl.querySelectorAll('.cell');
    cells.forEach(cell => {
        // Limpiar contenido de la celda antes de dibujar el estado actual
        cell.innerHTML = ''; 
        
        const r = parseInt(cell.getAttribute('data-row'));
        const c = parseInt(cell.getAttribute('data-col'));

        let elementToPlace = null;
        let elementType = '';

        // Chequear si hay una caja en esta posición
        const boxAtPos = estado.boxes.find(box => box.r === r && box.c === c);
        if (boxAtPos) {
            elementToPlace = document.createElement('div');
            elementToPlace.className = 'box';
            elementToPlace.innerHTML = '📦'; // Emoji para caja
            elementType = 'box';
        }

        // Chequear si el jugador está en esta posición
        const playerAtPos = estado.playerPos;
        if (r === playerAtPos.r && c === playerAtPos.c) {
            elementToPlace = document.createElement('div');
            elementToPlace.className = 'avatar';
            elementToPlace.innerHTML = '👤'; // Emoji para jugador
            elementType = 'player';
        }

        if (elementToPlace) {
             // Si hay caja y el jugador está en la misma posición, el jugador tiene prioridad visualmente.
             // Pero dado que las cajas no pueden superponerse con el avatar, esto es seguro.
            cell.appendChild(elementToPlace);
        }
    });

    // 2. Actualizar HUD (F3.2)
    document.getElementById('nivel-display').textContent = estado.levelIndex + 1;
    document.getElementById('movimiento-contador').textContent = estado.moves;

    // 3. Verificar condición de victoria y actualizar mensaje
    const winMessageEl = document.getElementById('mensaje-juego');
    if (checkWinCondition()) {
        estado.isGameOver = true;
        winMessageEl.className = 'nes-message';
        winMessageEl.style.backgroundColor = '#28a745'; // Verde éxito
        winMessageEl.innerHTML = `🎉 ¡Nivel ${estado.levelIndex + 1} superado! 🎉<br>Movimientos: ${estado.moves}.`;
        winMessageEl.classList.remove('hidden');
    } else if (estado.isGameOver) {
         // Si el juego terminó por otro motivo (ej. error de lógica), mostrar mensaje de fallo aquí.
    } else {
        winMessageEl.className = 'nes-message hidden';
    }

    // Habilitar/deshabilitar botones según el estado
    document.getElementById('btn-proximo-nivel').disabled = !estado.isGameOver;
    document.getElementById('btn-reiniciar').disabled = false; // Siempre activo para reiniciar el nivel actual
}


/**
 * Maneja la lógica de avance al siguiente nivel (F4.7).
 */
function nextLevel() {
    if (estado.levelIndex >= estado.maxLevel - 1) {
        alert("¡Has completado todos los niveles! ¡Excelente trabajo!");
        return;
    }

    const nextIndex = estado.levelIndex + 1;
    resetEstadoNivel(nextIndex);
    dibujar();
}


/**
 * Maneja la lógica de reinicio del nivel actual (F4.4).
 */
function restartLevel() {
    if (!confirm("¿Estás seguro de que quieres reiniciar el nivel? Se perderán los movimientos actuales.")) {
        return;
    }
    resetEstadoNivel(estado.levelIndex);
    dibujar();
}


/**
 * Maneja la entrada del teclado (R4, R5).
 */
function handleKeyDown(event) {
    if (estado.isGameOver) return;

    let direction = -1; // 0:Arriba, 1:Derecha, 2:Abajo, 3:Izquierda
    
    switch (event.key) {
        case 'ArrowUp':
            direction = 0; break;
        case 'ArrowDown':
            direction = 2; break;
        case 'ArrowLeft':
            direction = 3; break;
        case 'ArrowRight':
            direction = 1; break;
        default:
            return; // Ignorar otras teclas
    }

    event.preventDefault(); // Prevenir scroll del navegador

    // Intentar mover y actualizar el estado si es exitoso
    const success = attemptMove(direction);
    if (success) {
        dibujar(); // Redibujar solo si hubo un movimiento válido
    } else {
        console.log("Movimiento inválido o bloqueado.");
    }
}


/**
 * Inicialización principal de la aplicación (A3).
 */
function iniciar() {
    // 1. Configurar Listeners de Eventos
    document.addEventListener('keydown', handleKeyDown);

    document.getElementById('btn-reiniciar').addEventListener('click', restartLevel);
    document.getElementById('btn-proximo-nivel').addEventListener('click', nextLevel);

    // 2. Inicializar el estado y dibujar la primera escena (F4.5)
    if (resetEstadoNivel(estado.levelIndex)) {
        dibujar();
    } else {
        document.getElementById('mensaje-juego').innerHTML = "Error al cargar los niveles del juego.";
        document.getElementById('mensaje-juego').classList.remove('hidden');
    }
}

// Llamada de inicialización (A3)
iniciar();
