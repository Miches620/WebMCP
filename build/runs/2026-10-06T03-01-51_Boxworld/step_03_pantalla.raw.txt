### FILE: styles.css
```css
/* Estilos generales y de la fuente (NES/8bit) */
body {
    font-family: 'Press Start 2P', monospace; /* Usando una fuente que simule el estilo retro */
    background-color: #1a1a1a;
    color: #e0e0e0;
    display: flex;
    justify-content: center;
    align-items: flex-start;
    padding: 20px;
}

.container {
    width: 90%;
    max-width: 1000px;
    text-align: center;
}

h1 {
    color: #ffcc00; /* Color de título brillante */
    margin-bottom: 20px;
    font-size: 2em;
}

/* HUD (Head Up Display) */
.hud-container {
    display: flex;
    justify-content: space-around;
    margin-bottom: 15px;
    padding: 10px;
    border: 3px solid #444;
    background-color: #2c2c2c;
}

#nivel, #movimientos {
    font-size: 1.2em;
    color: #aaffaa; /* Color de información */
}

/* Tablero (Grid) */
.grid-board {
    display: grid;
    /* El JS establecerá las columnas dinámicamente, pero definimos un tamaño base */
    grid-template-columns: repeat(10, 1fr); /* Asumiendo 10 columnas por defecto */
    width: 100%;
    aspect-ratio: 1 / 1; /* Asegura que sea cuadrado si las celdas son iguales */
    border: 4px solid #8b4513; /* Borde de caja más grueso */
    background-color: #000;
}

/* Estilo base del casillero */
.casillero {
    width: 100%;
    height: 100%;
    box-sizing: border-box;
    border: 1px solid rgba(255, 255, 255, 0.1); /* Separación sutil */
    display: flex;
    justify-content: center;
    align-items: center;
}

/* Clases de tipo de casillero (Superposición visual) */

/* Piso base */
.piso {
    background-color: #3a3a3a; /* Gris oscuro para el piso */
}

/* Paredes (#) */
.pared {
    background-color: #8b4513; /* Marrón madera/tierra, color sólido de NES */
    border: 2px solid #6d370e;
    box-shadow: inset 0 0 5px rgba(0, 0, 0, 0.5);
}

/* Objetivo (.) - Se superpone sobre el piso/pared si es necesario, pero visualmente debe destacar */
.objetivo {
    background-color: #ffcc00; /* Amarillo brillante para objetivo */
    border: 2px dashed #e6b800;
    box-shadow: inset 0 0 10px rgba(255, 204, 0, 0.7);
}

/* Caja ($ o *) - Debe ser visible sobre el piso/objetivo */
.caja {
    background-color: #b8860b; /* Dorado/Marrón caja */
    border: 3px solid #a0522d;
    box-shadow: inset 0 0 10px rgba(0, 0, 0, 0.5);
}

/* Jugador (@) - Debe ser el elemento más visible y superponerse a todo */
.jugador {
    background-color: #ff4500; /* Naranja/Rojo brillante para el jugador */
    border: 3px solid #cc3700;
    box-shadow: 0 0 15px rgba(255, 69, 0, 0.8);
    z-index: 10; /* Asegura que esté por encima de todo */
}

/* Combinaciones (Superposición) */
/* Caja sobre Objetivo (*) */
.objetivo .caja {
    background-color: #b8860b;
    border: 3px solid #a0522d;
    box-shadow: inset 0 0 10px rgba(0, 0, 0, 0.5), 0 0 10px #ffcc00; /* Efecto de objetivo visible */
}

/* Mensaje y Controles */
.message-area {
    min-height: 30px;
    color: #aaffaa;
    font-size: 1.1em;
    margin-top: 20px;
    padding: 10px;
}

.controls button {
    background-color: #4CAF50; /* Verde NES */
    color: white;
    border: none;
    padding: 10px 20px;
    margin: 10px 10px 0 0;
    cursor: pointer;
    font-family: 'Press Start 2P', monospace;
    font-size: 0.9em;
    transition: background-color 0.2s;
}

.controls button:hover {
    background-color: #45a049;
}
```
