### FILE: js/controles.js
```javascript
let nivelActual = 0;
let estado;

/**
 * Inicializa el juego o reinicia un nivel específico.
 * @param {number} i - El índice del nivel a cargar.
 */
function iniciarNivel(i) {
    if (!NIVELES[i]) {
        document.getElementById('mensaje').textContent = "¡No hay más niveles!";
        return;
    }

    nivelActual = i;
    // 1. Crear el estado inicial del nivel
    estado = Reglas.crearEstado(NIVELES[i]);
    
    // 2. Limpiar mensajes y dibujar la escena
    document.getElementById('mensaje').textContent = "¡Mucha suerte! Usa las flechas para moverte.";
    dibujar(estado, nivelActual);
}

/**
 * Maneja el movimiento del jugador basado en las teclas presionadas.
 */
function manejarMovimientoTeclado(event) {
    let direccion;
    switch (event.key) {
        case 'ArrowUp':
            direccion = "arriba";
            break;
        case 'ArrowDown':
            direccion = "abajo";
            break;
        case 'ArrowLeft':
            direccion = "izquierda";
            break;
        case 'ArrowRight':
            direccion = "derecha";
            break;
        default:
            return; // No es una flecha de movimiento
    }

    event.preventDefault();

    // 1. Calcular el nuevo estado después del intento de movimiento
    const nuevoEstado = Reglas.mover(estado, direccion);
    
    // Solo actualizamos si hubo un movimiento exitoso (o al menos se intentó)
    estado = nuevoEstado;

    // 2. Dibujar la nueva escena y actualizar el contador
    dibujar(estado, nivelActual);

    // 3. Verificar condición de victoria
    if (Reglas.ganado(estado)) {
        document.getElementById('mensaje').textContent = "¡Nivel completado! Presiona Siguiente Nivel.";
    } else if (nuevoEstado.movimientos > estado.movimientos) {
         // Solo actualizamos el mensaje si realmente hubo un movimiento exitoso
        document.getElementById('mensaje').textContent = `Movimiento realizado. ¡Sigue intentándolo!`;
    }
}

/**
 * Maneja la lógica del botón 'Reiniciar Nivel'.
 */
function manejarReiniciar() {
    // Reinicia el estado al inicio del nivel actual
    iniciarNivel(nivelActual);
    document.getElementById('mensaje').textContent = "¡Has reiniciado! Inténtalo de nuevo.";
}

/**
 * Maneja la lógica del botón 'Siguiente Nivel'.
 */
function manejarSiguiente() {
    const siguienteNivelIndex = nivelActual + 1;
    if (siguienteNivelIndex < NIVELES.length) {
        iniciarNivel(siguienteNivelIndex);
        document.getElementById('mensaje').textContent = "¡Avanzando al siguiente desafío!";
    } else {
        document.getElementById('mensaje').textContent = "¡Has completado todos los niveles! ¡Buen trabajo!";
        // Deshabilitar botón de siguiente nivel si no hay más contenido
        document.getElementById('btn-siguiente').disabled = true;
    }
}

// --- Inicialización del Juego ---

// 1. Listener para el teclado (Flechas)
document.addEventListener('keydown', manejarMovimientoTeclado);

// 2. Listeners para los botones de control
document.getElementById('btn-reiniciar').addEventListener('click', manejarReiniciar);
document.getElementById('btn-siguiente').addEventListener('click', manejarSiguiente);

// 3. Iniciar el juego en el Nivel 0
iniciarNivel(0);
```
