/* Estilos generales y estética NES/8bits */
body {

    padding: 20px;
}

h1 {
    text-align: center;
    color: #ffcc00; /* Color brillante para el título */
    margin-bottom: 15px;
    font-size: 2em;
}

/* HUD (Heads Up Display) */
.hud {
    display: flex;
    justify-content: space-between;
    margin-bottom: 15px;
    padding: 10px;
    background-color: #444;
    border: 3px solid #222;
}

.info-box {
    font-size: 1.2em;

    background-color: #666;
    border: 2px dashed #aaa;
}

/* Tablero de juego */
.game-board {
    display: grid;
    grid-template-columns: repeat(10, 1fr); /* Fijo a 10 columnas */
    width: 600px; /* Ancho fijo para mantener la proporción */
    height: 600px; /* Alto fijo para mantener la proporción */
    border: 4px solid #000;
    background-color: #222;
}

/* Estilo base del casillero */
.casillero {
    width: 100%;
    height: 100%;
    box-sizing: border-box;
    border: 1px solid rgba(0, 0, 0, 0.2); /* Separación sutil entre casilleros */
}

/* Definición de tipos de terreno/objetos (Clases) */

/* Piso (Espacio vacío normal) */
.casillero.piso {
    background-color: #8b4513; /* Marrón tierra */
}

/* Pared (#) */
.casillero.pared {
    background-color: #000080; /* Azul oscuro, sólido */
    border: 2px solid #0000ff;
    box-shadow: inset 0 0 5px rgba(0, 0, 255, 0.5);
}

/* Objetivo (.) */
.casillero.objetivo {
    background-color: #38761d; /* Verde objetivo */
    border: 2px dashed #6aa84f;
}

/* Caja ($ o *) */
.casillero.caja {
    background-color: #cc9900; /* Amarillo/Dorado para cajas */
    box-shadow: inset 0 0 10px rgba(255, 255, 0, 0.7);
}

/* Jugador (@) - Se superpone sobre el piso o objetivo */
.casillero.jugador {
    background-color: #ff4500; /* Naranja brillante para el jugador */
    border: 3px solid #ff8c00;
    box-shadow: inset 0 0 15px rgba(255, 69, 0, 0.9);
}

/* Combinaciones de clases (Prioridad visual) */

/* Caja sobre Objetivo (*): La caja debe ser visible y el objetivo también */
.casillero.objetivo.caja {
    background-color: #cc9900; /* Caja domina, pero mantiene un toque verde */
    border: 2px solid #38761d;
}

/* Jugador sobre Objetivo (Si el jugador está en un objetivo) */
.casillero.objetivo.jugador {
    background-color: #ff4500; /* El jugador es lo más visible */
    border: 3px solid #ff8c00;
}

/* Jugador sobre Caja (No debería pasar por reglas, pero si pasa) */
.casillero.caja.jugador {
    background-color: #ff4500;
    border: 3px solid #ff8c00;
}


/* Mensaje de estado */
.message-area {
    min-height: 40px;
    padding: 10px;
    margin-top: 15px;
    background-color: #2a2a2a;
    border: 3px solid #00ff00; /* Borde de éxito/mensaje */
    color: #00ff00;
}

/* Controles y Botones */
.controls {
    display: flex;
    justify-content: space-between;
    margin-top: 20px;
}

button {

    transition: all 0.1s ease;
}

button:hover:not(:disabled) {
    background-color: #ffdd55;
    box-shadow: 6px 6px 0 #b38f00;
}

button:active:not(:disabled) {
    transform: translate(2px, 2px);
    box-shadow: 2px 2px 0 #b38f00;
}

button:disabled {
    background-color: #666;
    cursor: not-allowed;
    box-shadow: 4px 4px 0 #333;
}
