crearEstado: function(nivel) {
        const filas = nivel.length;
        if (filas === 0) return null;
        const columnas = nivel[0].length;

        let jugadorPos = { fila: -1, col: -1 };
        let cajasList = [];
        let objetivosList = [];
        let mapaGrid = [];

        for (let r = 0; r < filas; r++) {
            let currentRowChars = [];
            for (let c = 0; c < columnas; c++) {
                const char = nivel[r][c];
                let mapChar = ' '; // Default for non-fixed elements

                // Handle fixed/objective markers first
                if (char === '.') {
                    objetivosList.push({ fila: r, col: c });
                    mapChar = '.';
                } 
                
                // Handle movable objects and players
                if (char === '@' || char === '+') {
                    jugadorPos = { fila: r, col: c };
                    mapChar = ' '; // Player occupies empty space in the base map
                } else if (char === '$' || char === '*') {
                    cajasList.push({ fila: r, col: c });
                    mapChar = ' '; // Box occupies empty space in the base map
                } 
                
                // Handle objective markers that are also objects/players (* or +)
                if (char === '*' || char === '+') {
                     // These symbols mark both an object location and an objective.
                     objetivosList.push({ fila: r, col: c });
                     mapChar = ' '; // Map remains empty space
                }

                // Handle walls
                if (char === '#') {
                    mapChar = '#';
                } 
                
                currentRowChars.push(mapChar);
            }
            // Fix: mapa must be an array of strings, not an array of character arrays
            mapaGrid[r] = currentRowChars.join('');
        }

        return {
            mapa: mapaGrid, // Array of strings
            jugador: jugadorPos,
            cajas: cajasList,
            objetivos: objetivosList,
            movimientos: 0
        };
    },
