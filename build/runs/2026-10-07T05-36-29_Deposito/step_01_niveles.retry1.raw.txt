const NIVELES = [
    // Nivel 1: Fácil (2 cajas)
    [
        "########",
        "#@$ .#   #", // R1: @(1,1), $(1,2), .(1,4). Space at (1,3) is empty.
        "#   #.#   #", // R2: Objective at (2,3)
        "## #.$ # #", // R3: Box $ at (3,5)
        "#####*###", // R4: Box * at (4,5). Objective at (4,4) is implied by the structure. Let's adjust to fit 8 chars and clear definition.
        "#       .#", // R5: Empty row for spacing/pathing
        "#   . . # #"  // R6: Objectives at (6,3), (6,4)
    ],
    // Nivel 2: Medio (3 cajas)
    [
        "########",
        "#@.#...#", // R1: Player @(1,1). Objective .(1,3)

        "## #.$###", // R3: Box $(3,3). Objective at (3,4) is implied.
        "#.#*...#", // R4: Box *(4,3). Objective at (4,1) is implied.
        "#.....#.#", // R5: Pathing row
        "########"
    ],
    // Nivel 3: Difícil (4 cajas)
    [
        "########",

        "# .# # # . #", // R4: Box/Objective interaction. Objective at (4,6)
