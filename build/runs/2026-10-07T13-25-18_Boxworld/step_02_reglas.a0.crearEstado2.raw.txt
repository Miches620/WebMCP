createEstado: function(nivel) {
    const filas = nivel.length;
    if (filas === 0) return null;
    const columnas = nivel[0].length;

    let jugadorPos = { fila: -1, col: -1 };
    let cajasList = [];
    let objetivosList = [];
    let mapaGridStrings = [];

    for (let r = 0; r < filas; r++) {
        let currentRowMap = '';

        for (let c = 0; c < columnas; c++) {
            const char = nivel[r][c];
            let mapChar = ' '; // Default character for the base map

            // 1. Detect Player Position (@)
            if (char === '@') {
                jugadorPos = { fila: r, col: c };
                mapChar = ' ';
            } 
            // 2. Detect Walls (#)
            else if (char === '#') {
                mapChar = '#';
            } 
            // 3. Detect Boxes ($ or *)
            else if (char === '$' || char === '*') {
                cajasList.push({ fila: r, col: c });
                mapChar = ' '; // Box location is empty space in the base map
            } 
            // 4. Detect Objectives (., +, *)
            else if (char === '.' || char === '+' || char === '*') {
                // Add to objectives list if not already present
                if (!objetivosList.some(obj => obj.fila === r && obj.col === c)) {
                    objetivosList.push({ fila: r, col: c });
                }
                mapChar = '.'; // Objective spots are marked with '.' in the base map
            } 
            // 5. Handle empty space or other characters
            else {
                mapChar = ' ';
            }

            currentRowMap += mapChar;
        }
        mapaGridStrings.push(currentRowMap);
    }

    return {
        mapa: mapaGridStrings, // Array of strings
        jugador: jugadorPos,
        cajas: cajasList,
        objetivos: objetivosList,
        movimientos: 0
    };
}
