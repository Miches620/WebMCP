crearEstado(nivel) {
    let jugador = null;
    let cajas = [];
    let objetivos = [];
    let mapaLimpio = [];

    for (let r = 0; r < nivel.length; r++) {
        let filaString = '';
        for (let c = 0; c < nivel[r].length; c++) {
            const char = nivel[r][c];
            let cleanChar = ' '; // Default fallback

            // --- Object Detection ---
            let isPlayerStart = false;
            if (char === '@' || char === '+') {
                jugador = { fila: r, col: c };
                isPlayerStart = true;
            } 
            
            if (char === '$' || char === '*') {
                cajas.push({ fila: r, col: c });
            } 
            
            // Objectives are at ., *, or +
            if (char === '.' || char === '*' || char === '+') {
                objetivos.push({ fila: r, col: c });
            }

            // --- Map Cleaning Logic ---
            if (char === '#') {
                cleanChar = '#';
            } else if (isPlayerStart) { 
                // Player start location must be treated as open floor space in the map.
                cleanChar = ' ';
            } else if (char === '.') {
                cleanChar = '.'; // Objective spot remains '.'
            } else if (char === '*' || char === '+') {
                 // If it's an objective marker but not player start, it becomes floor space for the map.
                cleanChar = '.'; 
            } else if (char === ' ') {
                cleanChar = ' '; // Empty floor space
            } else if (char === '$') {
                // Box location is treated as open space (' ') in the map.
                cleanChar = ' '; 
            } else {
                 // Fallback for safety
                 cleanChar = ' ';
            }

            filaString += cleanChar;
        }
        mapaLimpio.push(filaString);
    }

    return {
        mapa: mapaLimpio,
        jugador: jugador,
        cajas: cajas,
        objetivos: objetivos,
        movimientos: 0
    };
}
