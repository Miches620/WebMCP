crearEstado: function(nivel) {
    let jugador = null;
    let cajas = [];
    let objetivos = [];
    let mapa = [];

    for (let r = 0; r < nivel.length; r++) {
      let filaActual = "";
      for (let c = 0; c < nivel[r].length; c++) {
        const char = nivel[r][c];
        let mapChar = " ";

        // 1. Track Player Location
        if (char === '@' || char === '+') {
          jugador = { fila: r, col: c };
        }

        // 2. Track Boxes
        if (char === '$' || char === '*') {
          cajas.push({ fila: r, col: c });
        }

        // 3. Track Objectives
        if (char === '.' || char === '*' || char === '+') {
          objetivos.push({ fila: r, col: c });
        }

        // 4. Determine Map Character (Priority: Wall > Objective > Floor)
        if (char === '#') {
          mapChar = "#";
        } else if (['.', '*', '+'].includes(char)) {
          // If it's an objective marker, the map shows '.'
          mapChar = ".";
        } else if (char === '$' || char === '@' || char === ' ') {
            // Boxes ($) or Player start (@) contribute a space to the visible map 
            // unless they are objectives.
            mapChar = " ";
        } else {
          // Default case for empty floor space
          mapChar = " ";
        }

        filaActual += mapChar;
      }
      mapa.push(filaActual);
    }

    return {
      mapa: mapa,
      jugador: jugador,
      cajas: cajas,
      objetivos: objetivos,
      movimientos: 0
    };
  },
