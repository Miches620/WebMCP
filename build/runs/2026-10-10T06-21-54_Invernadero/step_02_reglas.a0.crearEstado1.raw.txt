crearEstado(nivel) {
        const filas = nivel.length;
        if (filas === 0) return { mapa: [], jugador: null, cajas: [], objetivos: [], movimientos: 0 };
        const columnas = nivel[0].length;

        let jugadorInicial = null;
        let cajasIniciales = [];
        let objetivosIniciales = [];

        // Paso 1: Identificar posiciones iniciales y objetivos
        for (let r = 0; r < filas; r++) {
            for (let c = 0; c < columnas; c++) {
                const char = nivel[r][c];
                if (char === '@' || char === '+') {
                    // Jugador puede empezar en @ o +
                    jugadorInicial = { fila: r, col: c };
                } else if (char === '$' || char === '*') {
                    // Las cajas se identifican por $ o *
                    cajasIniciales.push({ fila: r, col: c });
                } 
                
                // Identificar objetivos: '.' es el marcador estándar. 
                // Para cubrir los fallos reportados sobre '*' y '+', asumimos que estos también pueden marcar un objetivo si están presentes en la posición de inicio/caja.
                if (char === '.') {
                    objetivosIniciales.push({ fila: r, col: c });
                } else if (char === '*') {
                    // Si * marca una caja sobre objetivo, el objetivo sigue existiendo aquí.
                    objetivosIniciales.push({ fila: r, col: c });
                } else if (char === '+') {
                     // Si + es usado para jugador en objetivo
                     objetivosIniciales.push({ fila: r, col: c });
                }
            }
        }

        // Paso 2: Construir el mapa limpio (solo #, ' ', '.')
        let mapaLimpio = [];
        for (let r = 0; r < filas; r++) {
            let filaMapa = '';
            for (let c = 0; c < columnas; c++) {
                const char = nivel[r][c];
                if (char === '#') {
                    filaMapa += '#'; // Pared
                } else if (char === '.') {
                    // Objetivo, en el mapa limpio es piso '.'
                    filaMapa += '.'; 
                } else if (char === '@' || char === '$' || char === '*') {
                    // Jugador o caja en el mapa limpio son piso ' '
                    filaMapa += ' '; 
                } else {
                    // Cualquier otro carácter (incluyendo '+') se trata como espacio vacío para el mapa.
                    filaMapa += ' ';
                }
            }
            mapaLimpio.push(filaMapa);
        }

        return {
            mapa: mapaLimpio,
            jugador: jugadorInicial,
            cajas: cajasIniciales,
            objetivos: objetivosIniciales,
            movimientos: 0
        };
    }
