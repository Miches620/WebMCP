crearEstado(nivel) {
    let jugador = null;
    let cajas = [];
    let objetivos = [];
    let nuevoMapa = [];

    for (let r = 0; r < nivel.length; r++) {
        let rowChars = [];
        for (let c = 0; c < nivel[r].length; c++) {
            const char = nivel[r][c];
            
            // 1. Detect object locations and objectives
            if (char === '@') {
                jugador = { fila: r, col: c };
            } else if (char === '$' || char === '*') {
                cajas.push({ fila: r, col: c });
            } 
            // Objectives are marked by '.', '+', or '*'
            if (char === '.' || char === '+' || char === '*') {
                 objetivos.push({ fila: r, col: c });
            }

            // 2. Build the clean map layer character by character
            let mapChar;
            if (char === '#') {
                mapChar = '#';
            } else if (char === '.' || char === '+' || char === '*') {
                // If it was an objective marker, the map shows '.'
                mapChar = '.';
            } else if (char === ' ') {
                mapChar = ' ';
            } else {
                // Handles '@' and '$'. They occupy floor space. 
                mapChar = ' ';
            }
            rowChars.push(mapChar);
        }
        nuevoMapa.push(rowChars.join(''));
    }

    return {
        mapa: nuevoMapa, 
        jugador: jugador,
        cajas: cajas,
        objetivos: objetivos,
        movimientos: 0
    };
}
