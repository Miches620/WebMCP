### FILE: styles.css
```css
/* Paleta de colores cálidos */
:root {
    --color-pared: #8b4513; /* Marrón oscuro (SaddleBrown) */
    --color-piso: #d2b48c; /* Crema/Beige claro (Tan) */
    --color-objetivo: #a0522d; /* Terracota más profundo (Sienna) */
    --color-caja: #6b8e23; /* Verde oliva (OliveDrab) */
    --color-jugador: #ff7f44; /* Naranja/Terracota brillante para el jugador */
    --color-fondo: #f5f5dc; /* Beige muy claro (Beige) */
}

body {
    font-family: 'Arial', sans-serif;
    background-color: var(--color-fondo);
    display: flex;
    justify-content: center;
    padding: 20px;
    margin: 0;
}

.contenedor-juego {
    /* Usamos un ancho máximo para que el contenido no se estire demasiado, pero permitimos que los elementos internos definan su flujo */
    max-width: 95%; 
    text-align: center;
    background-color: #fff8e1; /* Fondo ligeramente más oscuro para el contenedor principal */
    padding: 20px;
    border-radius: 10px;
    box-shadow: 0 4px 15px rgba(0, 0, 0, 0.1);
}

h1 {
    color: #8b4513; /* Color de título marrón */
    margin-bottom: 20px;
}

/* HUD (Head Up Display) */
.hud {
    display: flex;
    justify-content: space-around;
    margin-bottom: 20px;
    padding: 10px;
    background-color: #e9d8c3; /* Fondo suave para el HUD */
    border-radius: 5px;
}

#nivel, #movimientos {
    font-size: 1.2em;
    font-weight: bold;
    color: #5a4030;
}

/* Tablero de juego (Grid) */
.grid-container {
    display: grid;
    /* El ancho del contenedor debe ser determinado por el contenido fijo de los casilleros, no por el texto circundante */
    border: 2px solid var(--color-pared);
    background-color: #e0d5c8; /* Fondo del tablero ligeramente diferente */
    margin: 10px auto;
    /* Aseguramos que el grid se ajuste al contenido de sus hijos (los casilleros) */
    width: max-content; 
}

/* Estilo base de cada casillero */
.casillero {
    width: 40px;
    height: 40px;
    box-sizing: border-box;
    display: flex;
    align-items: center;
    justify-content: center;
    font-size: 1em; /* Para asegurar que los emojis/iconos se vean bien */
}

/* Definición de clases por tipo de terreno/objeto */

/* Pared (#) */
.casillero.pared {
    background-color: var(--color-pared);
    border: 1px solid #6d380e;
    box-shadow: inset 0 0 5px rgba(0, 0, 0, 0.2);
}

/* Piso (' ') */
.casillero.piso {
    background-color: var(--color-piso);
    border: 1px solid #c3b09a;
}

/* Objetivo ('.') */
.casillero.objetivo {
    background-color: var(--color-objetivo);
    border: 2px dashed #7d4526;
    box-shadow: inset 0 0 8px rgba(160, 82, 45, 0.5);
}

/* Caja ('$') */
.casillero.caja {
    background-color: var(--color-caja);
    border: 2px solid #4a691e;
    box-shadow: inset 0 0 8px rgba(107, 142, 35, 0.7);
}

/* Jugador ('@') */
.casillero.jugador {
    background-color: var(--color-jugador);
    border: 3px solid #cc6633;
    box-shadow: 0 0 10px rgba(255, 127, 68, 0.8);
}

/* Combinaciones de clases (Ejemplo: Caja sobre Objetivo) */
.casillero.objetivo.caja {
    background-color: var(--color-caja); /* La caja domina el color */
    border: 2px solid #4a691e;
    box-shadow: inset 0 0 8px rgba(107, 142, 35, 0.7), 0 0 10px var(--color-objetivo); /* Efecto de superposición */
}

/* Mensaje y Controles */
.mensaje-area {
    min-height: 30px;
    padding: 10px;
    margin-top: 20px;
    background-color: #fffacd; /* Amarillo crema suave para el mensaje */
    border: 1px solid #d2b48c;
    border-radius: 5px;
    /* FIX: Limitar el ancho del mensaje para que no estire el contenedor principal */
    max-width: 90%; 
}

.controles-botones button {
    padding: 10px 20px;
    margin: 10px 10px 0 0;
    font-size: 1em;
    cursor: pointer;
    border: none;
    border-radius: 5px;
    transition: background-color 0.2s;
}

#btn-reiniciar {
    background-color: #d9534f; /* Rojo suave */
    color: white;
}

#btn-siguiente {
    background-color: #5cb85c; /* Verde éxito */
    color: white;
}

button:hover:not(:disabled) {
    opacity: 0.9;
}

/* Estilo para botones deshabilitados (aunque el JS debe encargarse de esto, es buena práctica) */
button:disabled {
    background-color: #cccccc;
    cursor: not-allowed;
}
```
