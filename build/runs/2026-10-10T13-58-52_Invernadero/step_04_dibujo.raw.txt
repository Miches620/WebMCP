```javascript
let nivelActual = 0;
let estado;

/**
 * Inicializa el juego en un nivel específico.
 * @param {number} i - Índice del nivel a cargar (0-based).
 */
function iniciarNivel(i) {
    nivelActual = i;
    // Crear el nuevo estado basado en los niveles definidos
    estado = Reglas.crearEstado(NIVELES[i]);

    // Limpiar mensajes y dibujar el tablero
    document.getElementById('mensaje').textContent = '';
    dibujar(estado, nivelActual + 1);
}

/**
 * Maneja la lógica de movimiento del jugador al presionar flechas.
 */
function manejarMovimientoTeclado(event) {
    let direccion;
    switch (event.key) {
        case 'ArrowUp':
            direccion = "arriba";
            break;
        case 'ArrowDown':
            direccion = "abajo";
            break;
        case 'ArrowLeft':
            direccion = "izquierda";
            break;
        case 'ArrowRight':
            direccion = "derecha";
            break;
        default:
            return; // No es una flecha de movimiento
    }

    event.preventDefault();

    // 1. Calcular el nuevo estado después del intento de movimiento
    const nuevoEstado = Reglas.mover(estado, direccion);

    // Actualizar el estado global solo si hubo un cambio (aunque Reglas.mover ya maneja esto)
    estado = nuevoEstado;

    // 2. Dibujar el tablero con el nuevo estado y verificar la victoria
    dibujar(estado, nivelActual + 1);

    if (Reglas.ganado(estado)) {
        document.getElementById('mensaje').textContent = "¡Nivel completado! ¡Buen trabajo!";
    } else if (nuevoEstado.movimientos > estado.movimientos) {
         // Solo mostrar mensaje si el movimiento fue exitoso y no es la primera vez en el nivel
        if (!document.getElementById('mensaje').textContent || document.getElementById('mensaje').textContent === "¡Empieza a moverte!") {
             document.getElementById('mensaje').textContent = "";
        }
    } else {
         // Si no se movió, limpiar mensaje si estaba mostrando algo de éxito
         if (document.getElementById('mensaje').textContent !== "" && !Reglas.ganado(estado)) {
            document.getElementById('mensaje').textContent = "Inténtalo de nuevo.";
        }
    }
}

// --- Listeners de Eventos ---

// 1. Listener para el teclado
document.addEventListener('keydown', manejarMovimientoTeclado);

// 2. Botón Reiniciar
document.getElementById('btn-reiniciar').addEventListener('click', () => {
    iniciarNivel(nivelActual);
});

// 3. Botón Siguiente Nivel
document.getElementById('btn-siguiente').addEventListener('click', () => {
    if (nivelActual < NIVELES.length - 1) {
        const siguienteNivel = nivelActual + 1;
        iniciarNivel(siguienteNivel);
    } else {
        document.getElementById('mensaje').textContent = "¡Has completado todos los niveles!";
    }
});

// Inicializar el juego al cargar la página
iniciarNivel(0);
```
